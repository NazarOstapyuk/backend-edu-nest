# Tariffs, Monthly Enrollments and Adaptation (Part 2) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Configurable tariffs with dated prices, monthly enrollments with per-child conditions and automatic monthly charges, and hourly adaptation visits, all charging into the Part 1 finance ledger; plus admin UI.

**Architecture:** A pure billing engine (`billing-engine.ts`, on top of pure date helpers `dates.ts`) computes periods and amounts; services load data, call the engine and write `Charge` + `BilledPeriod` rows in one transaction (idempotent through a unique `(enrollmentId, periodStart)`). A daily `@nestjs/schedule` cron and a manual endpoint run the same billing method. Frontend adds a "Тарифи" page and two tabs on the child finance card.

**Tech Stack:** NestJS 11, Prisma 7 (prisma-client-js, adapter-pg), `@nestjs/schedule@6.1.3`, class-validator, Jest. Frontend: React 19, MUI 9, Zustand 5, react-hook-form 7 + yup, axios.

**Spec:** `docs/superpowers/specs/2026-10-11-tariffs-design.md` (depends on Part 1: `docs/superpowers/specs/2026-10-11-finance-design.md`)

```ts
const DAY_MS = 24 * 60 * 60 * 1000;

const pad = (value: number) => String(value).padStart(2, '0');

export const toDay = (value: Date | string): Date => {
  const date = typeof value === 'string' ? new Date(value) : value;
  return new Date(Date.UTC(date.getUTCFullYear(), date.getUTCMonth(), date.getUTCDate()));
};

export const addDays = (day: Date, days: number): Date => new Date(day.getTime() + days * DAY_MS);

export const dayCount = (from: Date, to: Date): number =>
  Math.round((to.getTime() - from.getTime()) / DAY_MS) + 1;

export const daysInMonth = (year: number, month: number): number =>
  new Date(Date.UTC(year, month + 1, 0)).getUTCDate();

export const monthDay = (year: number, month: number, day: number): Date => {
  const normalized = new Date(Date.UTC(year, month, 1));
  const y = normalized.getUTCFullYear();
  const m = normalized.getUTCMonth();
  return new Date(Date.UTC(y, m, Math.min(day, daysInMonth(y, m))));
};

export const firstDayOfMonth = (day: Date): Date =>
  new Date(Date.UTC(day.getUTCFullYear(), day.getUTCMonth(), 1));

export const lastDayOfMonth = (day: Date): Date =>
  new Date(Date.UTC(day.getUTCFullYear(), day.getUTCMonth() + 1, 0));

export const maxDay = (a: Date, b: Date): Date => (a.getTime() >= b.getTime() ? a : b);

export const minDay = (a: Date, b: Date): Date => (a.getTime() <= b.getTime() ? a : b);

export const isoDay = (day: Date): string => day.toISOString().slice(0, 10);

export const uaDay = (day: Date): string =>
  `${pad(day.getUTCDate())}.${pad(day.getUTCMonth() + 1)}.${day.getUTCFullYear()}`;

export const overlaps = (aStart: Date, aEnd: Date | null, bStart: Date, bEnd: Date | null): boolean =>
  (bEnd === null || aStart.getTime() <= bEnd.getTime()) &&
  (aEnd === null || bStart.getTime() <= aEnd.getTime());

export const kyivToday = (now: Date = new Date()): Date => {
  const iso = new Intl.DateTimeFormat('en-CA', {
    timeZone: 'Europe/Kyiv',
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
  }).format(now);
  return new Date(`${iso}T00:00:00.000Z`);
};
```

```ts
import {

  EnrollmentInput,
  PeriodResult,
  computePeriod,
  dueBills,
  findPeriod,
  nextPeriod,
  periodsUntil,
  priceOn,
} from './billing-engine';
import { isoDay } from './dates';

const day = (value: string) => new Date(`${value}T00:00:00.000Z`);
const d = (value: string) => new Prisma.Decimal(value);

const input = (over: Partial<EnrollmentInput> = {}): EnrollmentInput => ({
  tariffName: 'Садочок',
  startDate: day('2026-10-15'),
  endDate: null,
  mode: BillingMode.anniversary,
  prices: [{ price: d('15500'), effectiveFrom: day('2026-01-01') }],
  adjustments: [],
  ...over,
});

const adj = (
  type: AdjustmentType,
  value: string | null,
  start: string,
  end: string | null,
  voided = false,
): AdjustmentLike => ({
  type,
  value: value === null ? null : d(value),
  startDate: day(start),
  endDate: end === null ? null : day(end),
  voidedAt: voided ? new Date() : null,
});

const ranges = (i: EnrollmentInput, until: string) =>
  periodsUntil(i, day(until)).map((r) => [isoDay(r.periodStart), isoDay(r.periodEnd), r.fullDays]);

const bill = (result: PeriodResult) => {
  if (!result.ok) throw new Error('expected a bill');
  return result.bill;
};

const amountFor = (i: EnrollmentInput, periodStart: string) => {
  const range = findPeriod(i, day(periodStart));
  if (!range) throw new Error('no period');
  return bill(computePeriod(i, range)).amount.toFixed(2);
};
```

```ts
import { AdjustmentType, BillingMode, Prisma } from '@prisma/client';
import { formatMoney } from '../finance/finance-calc';
import {
  addDays,
  dayCount,
  daysInMonth,
  firstDayOfMonth,
  isoDay,
  lastDayOfMonth,
  maxDay,
  minDay,
  monthDay,
  uaDay,
} from './dates';

export interface PriceLike {
  price: Prisma.Decimal;
  effectiveFrom: Date;
}

export interface AdjustmentLike {
  type: AdjustmentType;
  value: Prisma.Decimal | null;
  startDate: Date;
  endDate: Date | null;
  voidedAt: Date | null;
}

export interface EnrollmentInput {
  tariffName: string;
  startDate: Date;
  endDate: Date | null;
  mode: BillingMode;
  prices: PriceLike[];
  adjustments: AdjustmentLike[];
}

export interface PeriodRange {
  periodStart: Date;
  periodEnd: Date;
  fullDays: number;
  isFirst: boolean;
}

export interface PeriodBill {
  periodStart: Date;
  periodEnd: Date;
  amount: Prisma.Decimal;
  description: string;
  zero: boolean;
  note: string | null;
}

export type PeriodResult =
  | { ok: true; bill: PeriodBill }
  | { ok: false; periodStart: Date; periodEnd: Date; reason: 'no_price' };

const time = (day: Date) => day.getTime();

const isActiveOn = (adjustment: AdjustmentLike, day: Date) =>
  !adjustment.voidedAt &&
  time(adjustment.startDate) <= time(day) &&
  (adjustment.endDate === null || time(adjustment.endDate) >= time(day));

const findActive = (adjustments: AdjustmentLike[], type: AdjustmentType, day: Date) =>
  adjustments.find((a) => a.type === type && isActiveOn(a, day));

export function priceOn(prices: PriceLike[], day: Date): Prisma.Decimal | null {
  const candidates = prices
    .filter((p) => time(p.effectiveFrom) <= time(day))
    .sort((a, b) => time(b.effectiveFrom) - time(a.effectiveFrom));
  return candidates[0]?.price ?? null;
}

export function periodsUntil(input: EnrollmentInput, until: Date): PeriodRange[] {
  const { startDate, endDate, mode } = input;
  const year = startDate.getUTCFullYear();
  const month = startDate.getUTCMonth();
  const anchor = startDate.getUTCDate();
  const periods: PeriodRange[] = [];

  for (let k = 0; ; k++) {
    let periodStart: Date;
    let naturalEnd: Date;
    let fullDays: number;

    if (mode === BillingMode.anniversary) {
      periodStart = monthDay(year, month + k, anchor);
      naturalEnd = addDays(monthDay(year, month + k + 1, anchor), -1);
      fullDays = dayCount(periodStart, naturalEnd);
    } else {
      const monthStart = monthDay(year, month + k, 1);
      periodStart = k === 0 ? startDate : monthStart;
      naturalEnd = lastDayOfMonth(monthStart);
      fullDays = daysInMonth(monthStart.getUTCFullYear(), monthStart.getUTCMonth());
    }

    if (time(periodStart) > time(until)) break;
    if (endDate && time(periodStart) > time(endDate)) break;

    const periodEnd = endDate && time(endDate) < time(naturalEnd) ? endDate : naturalEnd;
    periods.push({ periodStart, periodEnd, fullDays, isFirst: k === 0 });
  }

  return periods;
}

export function computePeriod(input: EnrollmentInput, range: PeriodRange): PeriodResult {
  const { periodStart, periodEnd, fullDays, isFirst } = range;
  const custom = findActive(input.adjustments, AdjustmentType.custom_price, periodStart);
  const base = custom?.value ?? priceOn(input.prices, periodStart);
  if (!base) return { ok: false, periodStart, periodEnd, reason: 'no_price' };

  const percent = findActive(input.adjustments, AdjustmentType.discount_percent, periodStart);
  const fixed = findActive(input.adjustments, AdjustmentType.discount_amount, periodStart);

  let price = new Prisma.Decimal(base);
  if (percent?.value) price = price.mul(new Prisma.Decimal(100).minus(percent.value)).div(100);
  if (fixed?.value) price = price.minus(fixed.value);
  if (price.lt(0)) price = new Prisma.Decimal(0);

  const windowStart =
    isFirst && input.mode === BillingMode.calendar_full ? firstDayOfMonth(periodStart) : periodStart;
  const pausedDays = input.adjustments
    .filter((a) => a.type === AdjustmentType.pause && !a.voidedAt)
    .reduce((sum, a) => {
      const from = maxDay(a.startDate, windowStart);
      const to = minDay(a.endDate ?? periodEnd, periodEnd);
      return time(from) <= time(to) ? sum + dayCount(from, to) : sum;
    }, 0);
  const billableDays = Math.max(dayCount(windowStart, periodEnd) - pausedDays, 0);

  const amount = price
    .mul(billableDays)
    .div(fullDays)
    .toDecimalPlaces(2, Prisma.Decimal.ROUND_HALF_UP);

  const description =
    `${input.tariffName}, ${uaDay(periodStart)}–${uaDay(periodEnd)}: ${formatMoney(base)}` +
    (custom ? ' (індивідуальна ціна)' : '') +
    (percent?.value ? ` − ${percent.value.toString()}%` : '') +
    (fixed?.value ? ` − ${formatMoney(fixed.value)}` : '') +
    (billableDays < fullDays ? `, ${billableDays}/${fullDays} дн.` : '') +
    (pausedDays > 0 ? `, пауза ${pausedDays} дн.` : '');

  const zero = amount.isZero();
  const note = zero ? (billableDays === 0 && pausedDays > 0 ? 'pause' : 'zero amount') : null;

  return { ok: true, bill: { periodStart, periodEnd, amount, description, zero, note } };
}

export function dueBills(
  input: EnrollmentInput,
  billedStarts: Date[],
  today: Date,
  leadDays: number,
): PeriodResult[] {
  const billed = new Set(billedStarts.map(isoDay));
  return periodsUntil(input, addDays(today, leadDays))
    .filter((range) => !billed.has(isoDay(range.periodStart)))
    .map((range) => computePeriod(input, range));
}

export function nextPeriod(input: EnrollmentInput, billedStarts: Date[], today: Date): PeriodResult | null {
  const billed = new Set(billedStarts.map(isoDay));
  const horizon = addDays(maxDay(today, input.startDate), 400);
  const range = periodsUntil(input, horizon).find((r) => !billed.has(isoDay(r.periodStart)));
  return range ? computePeriod(input, range) : null;
}

export function findPeriod(input: EnrollmentInput, periodStart: Date): PeriodRange | null {
  return (
    periodsUntil(input, periodStart).find((range) => isoDay(range.periodStart) === isoDay(periodStart)) ?? null
  );
}
```

```ts
export const toTariffView = (tariff: LoadedTariff, today: Date) => {
  const current = priceOn(tariff.prices, today);
  return {
    id: tariff.id,
    name: tariff.name,
    category: tariff.category,
    unit: tariff.unit,
    packageSize: tariff.packageSize,

    if (!tariff) throw new NotFoundException('Tariff not found');
    return tariff;
  }

  private async assertNameFree(name: string) {
    const existing = await this.prisma.tariff.findUnique({ where: { name } });
    if (existing) throw new BadRequestException('Tariff with this name already exists');
  }

  async list(today: Date = kyivToday()) {
    const tariffs = await this.prisma.tariff.findMany({ include: tariffInclude, orderBy: { name: 'asc' } });
    return tariffs.map((t) => toTariffView(t, today));
  }

  async create(dto: CreateTariffDto, userId: string) {
    if (dto.unit === TariffUnit.package && !dto.packageSize) {
      throw new BadRequestException('Package size is required for package tariffs');
    }
    if (money(dto.price).lte(0)) throw new BadRequestException('Amount must be greater than 0');
    await this.assertNameFree(dto.name);

    const tariff = await this.prisma.tariff.create({
      data: {
        name: dto.name,
        category: dto.category,
        unit: dto.unit,
        packageSize: dto.unit === TariffUnit.package ? (dto.packageSize ?? null) : null,
        billingMode: dto.billingMode,
        prices: {
          create: { price: dto.price, effectiveFrom: toDay(dto.effectiveFrom), createdById: userId },
        },
      },
      include: tariffInclude,
    });
    return toTariffView(tariff, kyivToday());
  }

  async update(id: string, dto: UpdateTariffDto) {
    const tariff = await this.get(id);
    if (dto.name !== undefined && dto.name !== tariff.name) await this.assertNameFree(dto.name);

    const updated = await this.prisma.tariff.update({
      where: { id },
      data: { name: dto.name, billingMode: dto.billingMode, isActive: dto.isActive },
      include: tariffInclude,
    });
    return toTariffView(updated, kyivToday());
  }

  async addPrice(id: string, dto: AddTariffPriceDto, userId: string) {
    await this.get(id);
    if (money(dto.price).lte(0)) throw new BadRequestException('Amount must be greater than 0');
    const effectiveFrom = toDay(dto.effectiveFrom);

    const existing = await this.prisma.tariffPrice.findUnique({
      where: { tariffId_effectiveFrom: { tariffId: id, effectiveFrom } },
    });
    if (existing) throw new BadRequestException('Price for this date already exists');

    await this.prisma.tariffPrice.create({
      data: { tariffId: id, price: dto.price, effectiveFrom, createdById: userId },
    });
    return toTariffView(await this.get(id), kyivToday());
  }

  async delete(id: string) {
    await this.get(id);
    const [enrollments, visits] = await Promise.all([
      this.prisma.enrollment.count({ where: { tariffId: id } }),
      this.prisma.adaptationVisit.count({ where: { tariffId: id } }),
    ]);
    if (enrollments + visits > 0) throw new BadRequestException('Cannot delete: tariff is in use');

    await this.prisma.$transaction([
      this.prisma.tariffPrice.deleteMany({ where: { tariffId: id } }),
      this.prisma.tariff.delete({ where: { id } }),
    ]);
    return { message: 'Tariff deleted' };
  }
}
```

```ts
  category: e.tariff.category,
  amount: bill.amount,
  description: bill.description,
  periodStart: bill.periodStart,
  periodEnd: bill.periodEnd,
  sourceType: ChargeSource.monthly,
  sourceId: e.id,
  createdById: userId,
});

@Injectable()
export class BillingService {
  constructor(private prisma: PrismaService) {}

  async getLeadDays(): Promise<number> {
    const setting = await this.prisma.appSetting.findUnique({ where: { key: LEAD_KEY } });
    return setting ? Number(setting.value) : DEFAULT_LEAD_DAYS;
  }

  async setLeadDays(value: number) {
    if (!Number.isInteger(value) || value < 0 || value > 60) {
      throw new BadRequestException('Lead days must be between 0 and 60');
    }
    await this.prisma.appSetting.upsert({
      where: { key: LEAD_KEY },
      create: { key: LEAD_KEY, value: String(value) },
      update: { value: String(value) },
    });
    return { chargeLeadDays: value };
  }

  async billEnrollment(enrollmentId: string, userId: string, today: Date = kyivToday()) {
    const enrollment = await this.prisma.enrollment.findUnique({
      where: { id: enrollmentId },
      include: enrollmentInclude,
    });
    if (!enrollment) throw new NotFoundException('Enrollment not found');
    return this.billLoaded(enrollment, userId, today, await this.getLeadDays());
  }

  async runAll(userId: string, today: Date = kyivToday()) {
    const leadDays = await this.getLeadDays();
    const enrollments = await this.prisma.enrollment.findMany({ include: enrollmentInclude });
    const total = emptyResult();
    for (const enrollment of enrollments) {
      try {
        merge(total, await this.billLoaded(enrollment, userId, today, leadDays));
      } catch (error) {
        total.failed.push({
          enrollmentId: enrollment.id,
          message: error instanceof Error ? error.message : String(error),
        });
      }
    }
    return total;
  }

  async problems(today: Date = kyivToday()) {
    const leadDays = await this.getLeadDays();
    const enrollments = await this.prisma.enrollment.findMany({ include: enrollmentInclude });
    return enrollments.flatMap((e) =>
      dueBills(
        toEngineInput(e),
        e.billedPeriods.map((p) => p.periodStart),
        today,
        leadDays,
      ).flatMap((r) => (r.ok ? [] : [toProblem(e, r.periodStart)])),
    );
  }

  async recalculate(billedPeriodId: string, reason: string, userId: string) {
    const period = await this.prisma.billedPeriod.findUnique({
      where: { id: billedPeriodId },
      include: { enrollment: { include: enrollmentInclude } },
    });
    if (!period) throw new NotFoundException('Billed period not found');

    const enrollment = period.enrollment;
    const input = toEngineInput(enrollment);
    const range = findPeriod(input, period.periodStart);
    let bill: PeriodBill | null = null;
    if (range) {
      const result = computePeriod(input, range);
      if (!result.ok) throw new BadRequestException('No price for this date');
      bill = result.bill;
    }

    await this.prisma.$transaction(async (tx) => {
      if (period.chargeId) {
        const old = await tx.charge.findUnique({ where: { id: period.chargeId } });
        if (old && !old.voidedAt) await voidChargeInTx(tx, period.chargeId, reason, userId);
      }
      const charge =
        bill && !bill.zero ? await tx.charge.create({ data: chargeData(enrollment, bill, userId) }) : null;
      await tx.billedPeriod.update({
        where: { id: billedPeriodId },
        data: {
          periodEnd: bill ? bill.periodEnd : period.periodEnd,
          amount: bill ? bill.amount : new Prisma.Decimal(0),
          chargeId: charge ? charge.id : null,
          note: bill ? bill.note : 'after end date',
        },
      });
    });

    return { childId: enrollment.childId };
  }

  private async billLoaded(
    enrollment: LoadedEnrollment,
    userId: string,
    today: Date,
    leadDays: number,
  ): Promise<BillResult> {
    const result = emptyResult();
    const due = dueBills(
      toEngineInput(enrollment),
      enrollment.billedPeriods.map((p) => p.periodStart),
      today,
      leadDays,
    );

    for (const item of due) {
      if (!item.ok) {
        result.problems.push(toProblem(enrollment, item.periodStart));
        continue;
      }
      const bill = item.bill;
      try {
        await this.prisma.$transaction(async (tx) => {
          const charge = bill.zero
            ? null
            : await tx.charge.create({ data: chargeData(enrollment, bill, userId) });
          await tx.billedPeriod.create({
            data: {
              enrollmentId: enrollment.id,
              periodStart: bill.periodStart,
              periodEnd: bill.periodEnd,
              amount: bill.amount,
              chargeId: charge ? charge.id : null,
              note: bill.note,
            },
          });
        });
        if (bill.zero) result.skippedZero += 1;
        else result.created += 1;
      } catch (error) {
        if (!isUniqueViolation(error)) throw error;
        result.alreadyBilled += 1;
      }
    }

    return result;
  }
}
```

```ts
import { BadRequestException, Injectable, NotFoundException } from '@nestjs/common';
import { AdjustmentType, TariffUnit } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { money } from '../finance/finance-calc';
import { priceOn } from './billing-engine';
import { kyivToday, overlaps, toDay } from './dates';
import { CreateAdjustmentDto } from './dto/create-adjustment.dto';
import { CreateEnrollmentDto } from './dto/create-enrollment.dto';
import { EndEnrollmentDto } from './dto/end-enrollment.dto';
import { enrollmentInclude, toEnrollmentView } from './enrollment-view';

const validValue = (type: AdjustmentType, value: string | undefined) => {
  if (type === AdjustmentType.pause) return value === undefined;
  if (value === undefined) return false;
  const amount = money(value);
  if (type === AdjustmentType.discount_percent) return amount.gt(0) && amount.lte(100);
  return amount.gt(0);
};

@Injectable()
export class EnrollmentsService {
  constructor(private prisma: PrismaService) {}

  private async assertChild(childId: string) {
    const child = await this.prisma.child.findUnique({ where: { id: childId } });
    if (!child) throw new NotFoundException('Child not found');
  }

  private async getEnrollment(id: string) {
    const enrollment = await this.prisma.enrollment.findUnique({ where: { id }, include: enrollmentInclude });
    if (!enrollment) throw new NotFoundException('Enrollment not found');
    return enrollment;
  }

  async childIdOf(enrollmentId: string) {
    return (await this.getEnrollment(enrollmentId)).childId;
  }

  async listForChild(childId: string, today: Date = kyivToday()) {
    await this.assertChild(childId);
    const enrollments = await this.prisma.enrollment.findMany({
      where: { childId },
      include: enrollmentInclude,
      orderBy: { startDate: 'desc' },
    });
    return enrollments.map((e) => toEnrollmentView(e, today));
  }

  async create(childId: string, dto: CreateEnrollmentDto, userId: string) {
    await this.assertChild(childId);
    const tariff = await this.prisma.tariff.findUnique({ where: { id: dto.tariffId }, include: { prices: true } });
    if (!tariff) throw new NotFoundException('Tariff not found');
    if (!tariff.isActive) throw new BadRequestException('Tariff is not active');
    if (tariff.unit !== TariffUnit.month) throw new BadRequestException('Tariff unit must be month');

    const startDate = toDay(dto.startDate);
    const endDate = dto.endDate ? toDay(dto.endDate) : null;
    if (endDate && endDate.getTime() < startDate.getTime()) {
      throw new BadRequestException('End date must not be before start date');
    }
    if (!priceOn(tariff.prices, startDate)) throw new BadRequestException('No price for this date');

    const existing = await this.prisma.enrollment.findMany({ where: { childId, tariffId: dto.tariffId } });
    if (existing.some((e) => overlaps(startDate, endDate, e.startDate, e.endDate))) {
      throw new BadRequestException('Enrollment overlaps an existing enrollment on this tariff');
    }

    await this.prisma.enrollment.create({
      data: {
        childId,
        tariffId: dto.tariffId,
        startDate,
        endDate,
        billingMode: dto.billingMode ?? null,
        note: dto.note,
        createdById: userId,
      },
    });
    return this.listForChild(childId);
  }

  async end(id: string, dto: EndEnrollmentDto) {
    const enrollment = await this.getEnrollment(id);
    const endDate = toDay(dto.endDate);
    if (endDate.getTime() < enrollment.startDate.getTime()) {
      throw new BadRequestException('End date must not be before start date');
    }
    await this.prisma.enrollment.update({ where: { id }, data: { endDate } });
    return this.listForChild(enrollment.childId);
  }

  async addAdjustment(enrollmentId: string, dto: CreateAdjustmentDto, userId: string) {
    const enrollment = await this.getEnrollment(enrollmentId);
    const startDate = toDay(dto.startDate);
    const endDate = dto.endDate ? toDay(dto.endDate) : null;
    if (endDate && endDate.getTime() < startDate.getTime()) {
      throw new BadRequestException('End date must not be before start date');
    }
    if (!validValue(dto.type, dto.value)) throw new BadRequestException('Invalid adjustment value');

    const sameType = await this.prisma.enrollmentAdjustment.findMany({
      where: { enrollmentId, type: dto.type, voidedAt: null },
    });
    if (sameType.some((a) => overlaps(startDate, endDate, a.startDate, a.endDate))) {
      throw new BadRequestException('Adjustment overlaps an existing adjustment of the same type');
    }

    await this.prisma.enrollmentAdjustment.create({
      data: {
        enrollmentId,
        type: dto.type,
        value: dto.value,
        startDate,
        endDate,
        reason: dto.reason,
        createdById: userId,
      },
    });
    return this.listForChild(enrollment.childId);
  }

  async voidAdjustment(id: string, reason: string, userId: string) {
    const adjustment = await this.prisma.enrollmentAdjustment.findUnique({
      where: { id },
      include: { enrollment: { select: { childId: true } } },
    });
    if (!adjustment) throw new NotFoundException('Adjustment not found');
    if (adjustment.voidedAt) throw new BadRequestException('Adjustment is voided');

    await this.prisma.enrollmentAdjustment.update({
      where: { id },
      data: { voidedAt: new Date(), voidedById: userId, voidReason: reason },
    });
    return this.listForChild(adjustment.enrollment.childId);
  }
}
```

```ts
import { BadRequestException, Injectable, NotFoundException } from '@nestjs/common';
import { ChargeSource, Prisma, TariffUnit } from '@prisma/client';
import { randomUUID } from 'crypto';
import { PrismaService } from '../prisma/prisma.service';
import { activeAllocated, chargeState, formatMoney, money } from '../finance/finance-calc';
import { voidChargeInTx } from '../finance/void-charge';
import { priceOn } from './billing-engine';
import { toDay, uaDay } from './dates';
import { CreateAdaptationVisitDto } from './dto/create-adaptation-visit.dto';

const visitInclude = {
  tariff: { select: { id: true, name: true } },
  charge: { include: { allocations: true } },
} satisfies Prisma.AdaptationVisitInclude;

type LoadedVisit = Prisma.AdaptationVisitGetPayload<{ include: typeof visitInclude }>;

const toVisitView = (v: LoadedVisit) => ({
  id: v.id,
  tariff: v.tariff,
  date: v.date,
  hours: formatMoney(v.hours),
  ratePerHour: formatMoney(v.ratePerHour),
  amount: formatMoney(v.amount),
  note: v.note,
  chargeId: v.chargeId,
  chargeState: v.charge.voidedAt ? null : chargeState(v.charge.amount, activeAllocated(v.charge.allocations)),
  createdAt: v.createdAt,
  voidedAt: v.voidedAt,
  voidReason: v.voidReason,
});

@Injectable()
export class AdaptationService {
  constructor(private prisma: PrismaService) {}

  async listForChild(childId: string) {
    const child = await this.prisma.child.findUnique({ where: { id: childId } });
    if (!child) throw new NotFoundException('Child not found');
    const visits = await this.prisma.adaptationVisit.findMany({
      where: { childId },
      include: visitInclude,
      orderBy: { date: 'desc' },
    });
    return visits.map(toVisitView);
  }

  async create(childId: string, dto: CreateAdaptationVisitDto, userId: string) {
    const child = await this.prisma.child.findUnique({ where: { id: childId } });
    if (!child) throw new NotFoundException('Child not found');
    const tariff = await this.prisma.tariff.findUnique({ where: { id: dto.tariffId }, include: { prices: true } });
    if (!tariff) throw new NotFoundException('Tariff not found');
    if (tariff.unit !== TariffUnit.hour) throw new BadRequestException('Tariff unit must be hour');
    if (!tariff.isActive) throw new BadRequestException('Tariff is not active');

    const hours = money(dto.hours);
    if (hours.lte(0) || hours.gt(24)) throw new BadRequestException('Invalid hours');

    const date = toDay(dto.date);
    const rate = dto.ratePerHour ? money(dto.ratePerHour) : priceOn(tariff.prices, date);
    if (!rate) throw new BadRequestException('No price for this date');
    const amount = hours.mul(rate).toDecimalPlaces(2, Prisma.Decimal.ROUND_HALF_UP);
    if (amount.lte(0)) throw new BadRequestException('Amount must be greater than 0');

    const visitId = randomUUID();
    await this.prisma.$transaction(async (tx) => {
      const charge = await tx.charge.create({
        data: {
          childId,
          category: tariff.category,
          amount,
          description: `${tariff.name}, ${uaDay(date)}: ${formatMoney(hours)} год × ${formatMoney(rate)}`,
          periodStart: date,
          periodEnd: date,
          sourceType: ChargeSource.adaptation,
          sourceId: visitId,
          createdById: userId,
        },
      });
      await tx.adaptationVisit.create({
        data: {
          id: visitId,
          childId,
          tariffId: tariff.id,
          date,
          hours,
          ratePerHour: rate,
          amount,
          note: dto.note,
          chargeId: charge.id,
          createdById: userId,
        },
      });
    });
    return this.listForChild(childId);
  }

  async void(id: string, reason: string, userId: string) {
    const childId = await this.prisma.$transaction(async (tx) => {
      const visit = await tx.adaptationVisit.findUnique({ where: { id } });
      if (!visit) throw new NotFoundException('Adaptation visit not found');
      if (visit.voidedAt) throw new BadRequestException('Adaptation visit is voided');

      const charge = await tx.charge.findUnique({ where: { id: visit.chargeId } });
      if (charge && !charge.voidedAt) await voidChargeInTx(tx, visit.chargeId, reason, userId);

      await tx.adaptationVisit.update({
        where: { id },
        data: { voidedAt: new Date(), voidedById: userId, voidReason: reason },
      });
      return visit.childId;
    });
    return this.listForChild(childId);
  }
}
```

```ts
      const [{ data: tariffs }, { data: problems }, { data: settings }] = await Promise.all([
        tariffsApi.getList(),
        billingApi.getProblems(),
        billingApi.getSettings(),
      ]);
      set({ tariffs, problems, leadDays: settings.chargeLeadDays });
    } finally {
      set({ isLoading: false });
    }

    set({ isSaving: true });

    } finally {
      set({ isSaving: false });
```

```ts
import { create } from 'zustand';
import {
  enrollmentsApi,
  tariffsApi,
  type CreateAdjustmentDto,
  type CreateEnrollmentDto,
  type Enrollment,
  type Tariff,
} from '../../../shared/api';
import { useChildFinanceStore } from './childFinanceStore';

export type EnrollmentReasonTarget =
  | { kind: 'adjustment'; id: string; title: string }
  | { kind: 'recalculate'; id: string; title: string };

interface EnrollmentsState {
  childId: string | null;
  enrollments: Enrollment[];
  tariffOptions: Tariff[];
  isLoading: boolean;
  isSaving: boolean;
  createEnrollmentModal: { isOpen: boolean };
  endEnrollmentModal: { isOpen: boolean; enrollment: Enrollment | null };
  adjustmentModal: { isOpen: boolean; enrollment: Enrollment | null };
  reasonTarget: EnrollmentReasonTarget | null;
  setCreateEnrollmentModal: (value: Partial<{ isOpen: boolean }>) => void;
  setEndEnrollmentModal: (value: Partial<{ isOpen: boolean; enrollment: Enrollment | null }>) => void;
  setAdjustmentModal: (value: Partial<{ isOpen: boolean; enrollment: Enrollment | null }>) => void;
  setReasonTarget: (target: EnrollmentReasonTarget | null) => void;
  fetchEnrollments: (childId: string) => Promise<void>;
  createEnrollment: (dto: CreateEnrollmentDto) => Promise<void>;
  endEnrollment: (id: string, endDate: string) => Promise<void>;
  addAdjustment: (id: string, dto: CreateAdjustmentDto) => Promise<void>;
  submitReason: (reason: string) => Promise<void>;
  billNow: (id: string) => Promise<void>;
}

export const useEnrollmentsStore = create<EnrollmentsState>((set, get) => {
  const mutate = async (request: () => Promise<Enrollment[]>) => {
    set({ isSaving: true });
    try {
      const enrollments = await request();
      set({ enrollments });
      const { childId } = get();
      if (childId) await useChildFinanceStore.getState().fetchFinance(childId);
    } finally {
      set({ isSaving: false });
    }
  };

  const requireChild = () => {
    const { childId } = get();
    if (!childId) throw new Error('Child is not selected');
    return childId;
  };

  return {
    childId: null,
    enrollments: [],
    tariffOptions: [],
    isLoading: false,
    isSaving: false,
    createEnrollmentModal: { isOpen: false },
    endEnrollmentModal: { isOpen: false, enrollment: null },
    adjustmentModal: { isOpen: false, enrollment: null },
    reasonTarget: null,

    setCreateEnrollmentModal: (value) =>
      set((state) => ({ createEnrollmentModal: { ...state.createEnrollmentModal, ...value } })),

    setEndEnrollmentModal: (value) =>
      set((state) => ({ endEnrollmentModal: { ...state.endEnrollmentModal, ...value } })),

    setAdjustmentModal: (value) =>
      set((state) => ({ adjustmentModal: { ...state.adjustmentModal, ...value } })),

    setReasonTarget: (target) => set({ reasonTarget: target }),

    fetchEnrollments: async (childId) => {
      set({ isLoading: true, childId });
      try {
        const [{ data: enrollments }, { data: tariffs }] = await Promise.all([
          enrollmentsApi.getByChild(childId),
          tariffsApi.getList(),
        ]);
        set({ enrollments, tariffOptions: tariffs });
      } finally {
        set({ isLoading: false });
      }
    },

    createEnrollment: (dto) =>
      mutate(async () => (await enrollmentsApi.create(requireChild(), dto)).data),

    endEnrollment: (id, endDate) => mutate(async () => (await enrollmentsApi.end(id, { endDate })).data),

    addAdjustment: (id, dto) => mutate(async () => (await enrollmentsApi.addAdjustment(id, dto)).data),

    submitReason: async (reason) => {
      const target = get().reasonTarget;
      if (!target) return;
      await mutate(async () =>
        target.kind === 'adjustment'
          ? (await enrollmentsApi.voidAdjustment(target.id, reason)).data
          : (await enrollmentsApi.recalculate(target.id, reason)).data,
      );
    },

    billNow: (id) => mutate(async () => (await enrollmentsApi.bill(id)).data.enrollments),
  };
});
```

```ts
import { create } from 'zustand';
import {
  adaptationApi,
  tariffsApi,
  type AdaptationVisit,
  type CreateAdaptationVisitDto,
  type Tariff,
} from '../../../shared/api';
import { useChildFinanceStore } from './childFinanceStore';

interface AdaptationState {
  childId: string | null;
  visits: AdaptationVisit[];
  tariffOptions: Tariff[];
  isLoading: boolean;
  isSaving: boolean;
  createVisitModal: { isOpen: boolean };
  voidTarget: AdaptationVisit | null;
  setCreateVisitModal: (value: Partial<{ isOpen: boolean }>) => void;
  setVoidTarget: (visit: AdaptationVisit | null) => void;
  fetchVisits: (childId: string) => Promise<void>;
  createVisit: (dto: CreateAdaptationVisitDto) => Promise<void>;
  voidVisit: (reason: string) => Promise<void>;
}

export const useAdaptationStore = create<AdaptationState>((set, get) => {
  const mutate = async (request: () => Promise<AdaptationVisit[]>) => {
    set({ isSaving: true });
    try {
      const visits = await request();
      set({ visits });
      const { childId } = get();
      if (childId) await useChildFinanceStore.getState().fetchFinance(childId);
    } finally {
      set({ isSaving: false });
    }
  };

  return {
    childId: null,
    visits: [],
    tariffOptions: [],
    isLoading: false,
    isSaving: false,
    createVisitModal: { isOpen: false },
    voidTarget: null,

    setCreateVisitModal: (value) =>
      set((state) => ({ createVisitModal: { ...state.createVisitModal, ...value } })),

    setVoidTarget: (visit) => set({ voidTarget: visit }),

    fetchVisits: async (childId) => {
      set({ isLoading: true, childId });
      try {
        const [{ data: visits }, { data: tariffs }] = await Promise.all([
          adaptationApi.getByChild(childId),
          tariffsApi.getList(),
        ]);
        set({ visits, tariffOptions: tariffs });
      } finally {
        set({ isLoading: false });
      }
    },

    createVisit: (dto) =>
      mutate(async () => {
        const { childId } = get();
        if (!childId) throw new Error('Child is not selected');
        return (await adaptationApi.create(childId, dto)).data;
      }),

    voidVisit: async (reason) => {
      const target = get().voidTarget;
      if (!target) return;
      await mutate(async () => (await adaptationApi.void(target.id, reason)).data);
    },
  };
});
```

```tsx
        </Typography>

          <Button onClick={handleClose} color="inherit" sx={{ width: '100%' }}>
            Скасувати
          </Button>
          <Button onClick={handleSubmit(onSubmit)} variant="contained" disabled={isSaving} sx={{ width: '100%' }}>
            Зберегти
          </Button>
        </Box>
      </Box>
    </Dialog>
  )
}
```
