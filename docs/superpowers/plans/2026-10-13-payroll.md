# Teacher Payroll Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Admins keep versioned salary terms per teacher (monthly rate, per-lesson rates, recurring allowances); monthly payroll sheets are computed live from conducted lessons and the attendance calendar, approved into frozen snapshots, corrected automatically in later months, and paid out with advances; teachers see only their own payroll.

**Architecture:** New backend module `src/payroll/` with a pure `payroll-calc.ts` (working days, segments, base and allowance proration, rate lookup, lesson lines, corrections, totals), `SalaryTermsService` (versions with rates and allowances), `PayrollBuilder` (loads lessons, terms, calendar and booked lines; builds the `PayrollDetail` view for drafts and approved sheets) and `PayrollService` (head list/detail, teacher views, worked days, adjustments, payments, approve, reopen). Frontend adds a «Зарплати» admin page (tabs «Відомість», «Умови») and a teacher page «Моя зарплата».

**Tech Stack:** NestJS 11, Prisma 7, class-validator/class-transformer, Jest. Frontend: React 19, MUI 9, Zustand 5, react-hook-form 7 + yup, axios.

## Review Focus

1. A lesson recorded in September after September was approved must show up exactly once: as a `+` correction in the October draft, and after October is approved it must not reappear in the November draft. Pinned in Task 2 (`correctionLines`) and Task 4 (booked lines read only for dates before the month).
2. A teacher hired on 16 October with rate 33000 gets 16500.00 for October (11 of 22 working days), not the full rate; a public holiday in the kindergarten calendar lowers the norm only for terms with `calendarKind = kindergarten`. Pinned in Task 2.
3. Two admins approving two months of the same teacher at once cannot both book the same correction: approve locks its own sheet, then all approved sheets of the teacher; reopen locks the approved set first (id-ordered) so two reopens cannot deadlock. Pinned in Task 5 (lock order asserted).
4. Reopening September after October already booked a correction for a September lesson → 400 `Payroll has later corrections`; reopening with a payment made after approval → 400 `Payroll has payments after approval`. Pinned in Task 5.
5. A teacher calling `/me/payroll/2026-10` never sees who created payments or voided records. Pinned in Task 4.

import {
  GroupKind,
  PayrollAdjustmentKind,
  PayrollCalendar,
  PayrollLessonType,
  PayrollLineType,
  PayrollSourceType,
  Prisma,
} from '@prisma/client';
import { type CalendarEntryLike, dayInfo } from '../attendance/register-calc';
import {
  addDays,
  firstDayOfMonth,
  lastDayOfMonth,
  maxDay,
  minDay,
} from '../tariffs/dates';

type Decimal = Prisma.Decimal;

const ZERO = new Prisma.Decimal(0);

const round = (value: Decimal) =>
  value.toDecimalPlaces(2, Prisma.Decimal.ROUND_HALF_UP);

const time = (day: Date) => day.getTime();

export interface RateLike {
  lessonType: PayrollLessonType;
  subjectId: string | null;
  classGroupId: string | null;
  amount: Decimal;
}

export interface AllowanceLike {
  name: string;
  amount: Decimal;
}

export interface TermsLike {
  id: string;
  validFrom: Date;
  validTo: Date | null;
  monthlyRate: Decimal;
  calendarKind: PayrollCalendar;
  rates: RateLike[];
  allowances: AllowanceLike[];
}

export interface LessonSource {
  sourceType: PayrollSourceType;
  sourceId: string;
  date: Date;
  lessonType: PayrollLessonType;
  subjectId: string | null;
  classGroupId: string | null;
  title: string;
}

export interface BookedLine {
  sourceType: PayrollSourceType;
  sourceId: string;
  lessonType: PayrollLessonType;
  date: Date;
  title: string;
  amount: Decimal;
}

export interface AdjustmentLike {
  kind: PayrollAdjustmentKind;
  amount: Decimal;
  reason: string;
  voidedAt: Date | null;
}

export type LineMeta = {
  noRate?: boolean;
  normDays?: number;
  defaultDays?: number;
  workedDays?: number;
};

export interface ComputedLine {
  type: PayrollLineType;
  sourceType: PayrollSourceType | null;
  sourceId: string | null;
  lessonType: PayrollLessonType | null;
  date: Date | null;
  refMonth: Date | null;
  title: string;
  amount: Decimal;
  meta: LineMeta | null;
}

export type SheetDays = {
  normDays: number;
  defaultDays: number;
  workedDays: number;
  override: boolean;
};

export interface Segment {
  terms: TermsLike;
  days: number;
  norm: number;
}

export interface SheetInput {
  month: Date;
  terms: TermsLike[];
  entries: CalendarEntryLike[];
  workedDays: number | null;
  lessons: LessonSource[];
  past: LessonSource[];
  booked: BookedLine[];

  days: SheetDays;
  lines: ComputedLine[];
  warnings: { noRate: number; lessonsWithoutTerms: number };
  lessons: number;

const calendarGroup = (kind: PayrollCalendar) => {
  if (kind === PayrollCalendar.kindergarten)
    return { id: '', kind: GroupKind.kindergarten };
  if (kind === PayrollCalendar.school_class)
    return { id: '', kind: GroupKind.school_class };
  return null;
};

export function workingDays(
  from: Date,
  to: Date,
  kind: PayrollCalendar,
  entries: CalendarEntryLike[],
): number {
  const group = calendarGroup(kind);
  let count = 0;
  for (let day = from; time(day) <= time(to); day = addDays(day, 1)) {
    if (dayInfo(day, group, entries).working) count++;
  }
  return count;
}

export function termsOn(terms: TermsLike[], day: Date): TermsLike | null {
  return (
    terms.find(
      (t) =>
        time(t.validFrom) <= time(day) &&
        (t.validTo === null || time(day) <= time(t.validTo)),
    ) ?? null
  );
}

export function rateFor(
  terms: TermsLike,
  lesson: LessonSource,
): Decimal | null {
  const sameType = terms.rates.filter(
    (r) => r.lessonType === lesson.lessonType,
  );
  const specific = sameType.find(
    (r) =>
      (r.subjectId !== null && r.subjectId === lesson.subjectId) ||
      (r.classGroupId !== null && r.classGroupId === lesson.classGroupId),
  );
  const general = sameType.find(
    (r) => r.subjectId === null && r.classGroupId === null,
  );
  return (specific ?? general)?.amount ?? null;
}

export function monthSegments(
  month: Date,
  terms: TermsLike[],
  entries: CalendarEntryLike[],
): Segment[] {
  const from = firstDayOfMonth(month);
  const to = lastDayOfMonth(month);
  return terms
    .filter(
      (t) =>
        time(t.validFrom) <= time(to) &&
        (t.validTo === null || time(t.validTo) >= time(from)),
    )
    .sort((a, b) => time(a.validFrom) - time(b.validFrom))
    .map((t) => ({
      terms: t,
      days: workingDays(
        maxDay(from, t.validFrom),
        t.validTo ? minDay(to, t.validTo) : to,
        t.calendarKind,
        entries,
      ),
      norm: workingDays(from, to, t.calendarKind, entries),
    }));
}

export function sheetDays(
  segments: Segment[],
  override: number | null,
): SheetDays {
  const defaultDays = segments.reduce((sum, s) => sum + s.days, 0);
  const normDays = segments.length ? segments[segments.length - 1].norm : 0;
  return {
    normDays,
    defaultDays,
    workedDays: override ?? defaultDays,
    override: override !== null,
  };
}

const share = (amount: Decimal, segment: Segment, days: SheetDays) =>
  segment.norm === 0 || days.defaultDays === 0
    ? ZERO
    : amount
        .mul(segment.days)
        .div(segment.norm)
        .mul(days.workedDays)
        .div(days.defaultDays);

const plainLine = (
  type: PayrollLineType,
  title: string,
  amount: Decimal,
  meta: LineMeta | null = null,
): ComputedLine => ({
  type,
  sourceType: null,
  sourceId: null,
  lessonType: null,
  date: null,
  refMonth: null,
  title,
  amount,
  meta,
});

type LessonRef = Omit<LessonSource, 'subjectId' | 'classGroupId'>;

const lessonLine = (
  type: PayrollLineType,
  lesson: LessonRef,
  amount: Decimal,
  noRate: boolean,
  refMonth: Date | null,
): ComputedLine => ({
  type,
  sourceType: lesson.sourceType,
  sourceId: lesson.sourceId,
  lessonType: lesson.lessonType,
  date: lesson.date,
  refMonth,
  title: lesson.title,
  amount,
  meta: noRate ? { noRate: true } : null,
});

export function baseLines(
  segments: Segment[],
  days: SheetDays,
): ComputedLine[] {
  const lines: ComputedLine[] = [];
  if (segments.some((s) => s.terms.monthlyRate.gt(0))) {
    const amount = segments.reduce(
      (sum, s) => sum.add(share(s.terms.monthlyRate, s, days)),
      ZERO,
    );
    lines.push(
      plainLine(PayrollLineType.base, '', round(amount), {
        normDays: days.normDays,
        defaultDays: days.defaultDays,
        workedDays: days.workedDays,
      }),
    );
  }
  for (const segment of segments) {
    for (const allowance of segment.terms.allowances) {
      lines.push(
        plainLine(
          PayrollLineType.allowance,
          allowance.name,
          round(share(allowance.amount, segment, days)),
        ),
      );
    }
  }
  return lines;
}

const byDate = (a: LessonRef, b: LessonRef) =>
  time(a.date) - time(b.date) || a.sourceId.localeCompare(b.sourceId);

export function lessonLines(
  lessons: LessonSource[],
  terms: TermsLike[],
): { lines: ComputedLine[]; withoutTerms: number } {
  const lines: ComputedLine[] = [];
  let withoutTerms = 0;
  for (const lesson of [...lessons].sort(byDate)) {
    const version = termsOn(terms, lesson.date);
    if (!version) {
      withoutTerms++;
      continue;
    }
    const rate = rateFor(version, lesson);
    lines.push(
      lessonLine(
        PayrollLineType.lesson,
        lesson,
        rate ?? ZERO,
        rate === null,
        null,
      ),
    );
  }
  return { lines, withoutTerms };
}

const sourceKey = (item: LessonRef) => `${item.sourceType}:${item.sourceId}`;

export function correctionLines(
  past: LessonSource[],
  booked: BookedLine[],
  terms: TermsLike[],
): ComputedLine[] {
  const net = new Map<string, Decimal>();
  const original = new Map<string, Decimal>();
  const samples = new Map<string, BookedLine>();
  for (const line of booked) {
    const key = sourceKey(line);
    net.set(key, (net.get(key) ?? ZERO).add(line.amount));
    const top = original.get(key);
    if (top === undefined || line.amount.gt(top))
      original.set(key, line.amount);
    if (!samples.has(key)) samples.set(key, line);
  }
  const lines: ComputedLine[] = [];
  const counted = new Set<string>();
  for (const lesson of [...past].sort(byDate)) {
    const version = termsOn(terms, lesson.date);
    if (!version) continue;
    const key = sourceKey(lesson);
    counted.add(key);
    const refMonth = firstDayOfMonth(lesson.date);
    const bookedNet = net.get(key);
    if (bookedNet === undefined) {
      const rate = rateFor(version, lesson);
      lines.push(
        lessonLine(
          PayrollLineType.correction,
          lesson,
          rate ?? ZERO,
          rate === null,
          refMonth,
        ),
      );
      continue;
    }
    const top = original.get(key) ?? ZERO;
    const target = top.gt(0) ? top : ZERO;
    const diff = target.sub(bookedNet);
    if (!diff.isZero())
      lines.push(
        lessonLine(PayrollLineType.correction, lesson, diff, false, refMonth),
      );
  }
  const removed = [...samples.entries()]
    .filter(([key]) => !counted.has(key))
    .map(([, sample]) => sample)
    .sort(byDate);
  for (const sample of removed) {
    const bookedNet = net.get(sourceKey(sample)) ?? ZERO;
    if (bookedNet.isZero()) continue;
    lines.push(
      lessonLine(
        PayrollLineType.correction,
        sample,
        bookedNet.neg(),
        false,
        firstDayOfMonth(sample.date),
      ),
    );
  }
  return lines;
}

export function adjustmentLines(
  adjustments: AdjustmentLike[],
): ComputedLine[] {
  return adjustments
    .filter((a) => a.voidedAt === null)
    .map((a) =>
      plainLine(
        PayrollLineType.adjustment,
        a.reason,
        a.kind === PayrollAdjustmentKind.bonus ? a.amount : a.amount.neg(),
      ),
    );
}

export const sumAmounts = (items: { amount: Decimal }[]): Decimal =>
  items.reduce((sum, item) => sum.add(item.amount), ZERO);

export function computeSheet(input: SheetInput): ComputedSheet {
  const segments = monthSegments(input.month, input.terms, input.entries);
  const days = sheetDays(segments, input.workedDays);
  const lessons = lessonLines(input.lessons, input.terms);
  const lines = [
    ...baseLines(segments, days),
    ...lessons.lines,
    ...correctionLines(input.past, input.booked, input.terms),
    ...adjustmentLines(input.adjustments),
  ];
  return {
    days,
    lines,
    warnings: {
      noRate: lines.filter((l) => l.meta?.noRate).length,
      lessonsWithoutTerms: lessons.withoutTerms,
    },
    lessons: lessons.lines.length,
  };
}

import {
  BadRequestException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { ClassFormat, PayrollLessonType, Prisma, Role } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import type { AuthenticatedUser } from '../common/types/authenticated-user.type';
import { parseDay } from '../classes/access';
import { addDays, kyivToday } from '../tariffs/dates';
import {
  CloseSalaryTermsDto,
  CreateSalaryTermsDto,
  SalaryAllowanceDto,
  SalaryRateDto,
  UpdateSalaryTermsDto,
} from './dto/salary-terms.dto';
import { termsInclude, toTermsView } from './terms-view';

const positive = (value: string) => {
  const amount = new Prisma.Decimal(value);
  if (amount.lte(0))
    throw new BadRequestException('Amount must be greater than 0');
  return amount;
};

@Injectable()
export class SalaryTermsService {
  constructor(private prisma: PrismaService) {}

  async list(teacherId: string) {
    const teacher = await this.prisma.user.findUnique({
      where: { id: teacherId },
      select: { id: true, role: true },
    });
    if (!teacher || teacher.role !== Role.teacher)
      throw new NotFoundException('Teacher not found');
    const rows = await this.prisma.salaryTerms.findMany({
      where: { teacherId },
      include: termsInclude,
      orderBy: { validFrom: 'desc' },
    });
    const today = kyivToday();
    return rows.map((row) => toTermsView(row, today));
  }

  async get(id: string) {
    const row = await this.prisma.salaryTerms.findUnique({
      where: { id },
      include: termsInclude,
    });
    if (!row) throw new NotFoundException('Salary terms not found');
    return toTermsView(row, kyivToday());
  }

  async create(dto: CreateSalaryTermsDto, user: AuthenticatedUser) {
    const teacher = await this.prisma.user.findUnique({
      where: { id: dto.teacherId },
      select: { id: true, role: true },
    });
    if (!teacher) throw new NotFoundException('Teacher not found');
    if (teacher.role !== Role.teacher)
      throw new BadRequestException('User is not a teacher');
    const validFrom = parseDay(dto.validFrom);
    const rates = await this.validateRates(dto.rates);
    const allowances = this.validateAllowances(dto.allowances);
    const id = await this.prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT id FROM "User" WHERE id = ${teacher.id} FOR UPDATE`;
      const versions = await tx.salaryTerms.findMany({
        where: { teacherId: teacher.id },
        select: { id: true, validFrom: true, validTo: true },
      });
      const open = versions.find(
        (v) =>
          v.validTo === null && v.validFrom.getTime() < validFrom.getTime(),
      );
      const clash = versions.some(
        (v) =>
          v.id !== open?.id &&
          (v.validTo === null || v.validTo.getTime() >= validFrom.getTime()),
      );
      if (clash) throw new BadRequestException('Salary terms overlap');
      if (open)
        await tx.salaryTerms.update({
          where: { id: open.id },
          data: { validTo: addDays(validFrom, -1) },
        });
      const created = await tx.salaryTerms.create({
        data: {
          teacherId: teacher.id,
          validFrom,
          monthlyRate: new Prisma.Decimal(dto.monthlyRate),
          calendarKind: dto.calendarKind,
          note: dto.note?.trim() || null,
          createdById: user.id,
          rates: { create: rates },
          allowances: { create: allowances },
        },
        select: { id: true },
      });
      return created.id;
    });
    return this.get(id);
  }

  async update(id: string, dto: UpdateSalaryTermsDto) {
    await this.findNotStarted(this.prisma, id);
    const rates = dto.rates ? await this.validateRates(dto.rates) : null;
    const allowances = dto.allowances
      ? this.validateAllowances(dto.allowances)
      : null;
    await this.prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT id FROM "SalaryTerms" WHERE id = ${id} FOR UPDATE`;
      await this.findNotStarted(tx, id);
      if (rates) {
        await tx.salaryRate.deleteMany({ where: { termsId: id } });
        await tx.salaryRate.createMany({
          data: rates.map((rate) => ({ termsId: id, ...rate })),
        });
      }
      if (allowances) {
        await tx.salaryAllowance.deleteMany({ where: { termsId: id } });
        await tx.salaryAllowance.createMany({
          data: allowances.map((allowance) => ({ termsId: id, ...allowance })),
        });
      }
      await tx.salaryTerms.update({
        where: { id },
        data: {
          monthlyRate:
            dto.monthlyRate === undefined
              ? undefined
              : new Prisma.Decimal(dto.monthlyRate),
          calendarKind: dto.calendarKind,
          note: dto.note === undefined ? undefined : dto.note.trim() || null,
        },
      });
    });
    return this.get(id);
  }

  async remove(id: string) {
    await this.findNotStarted(this.prisma, id);
    await this.prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT id FROM "SalaryTerms" WHERE id = ${id} FOR UPDATE`;
      const row = await this.findNotStarted(tx, id);
      await tx.salaryTerms.updateMany({
        where: {
          teacherId: row.teacherId,
          validTo: addDays(row.validFrom, -1),
        },
        data: { validTo: row.validTo },
      });
      await tx.salaryTerms.delete({ where: { id } });
    });
    return { message: 'Salary terms deleted' };
  }

  async close(id: string, dto: CloseSalaryTermsDto) {
    const row = await this.prisma.salaryTerms.findUnique({
      where: { id },
      select: { validFrom: true, validTo: true },
    });
    if (!row) throw new NotFoundException('Salary terms not found');
    if (row.validTo !== null)
      throw new BadRequestException('Salary terms are closed');
    const validTo = parseDay(dto.validTo);
    if (validTo.getTime() < row.validFrom.getTime())
      throw new BadRequestException('Invalid date range');
    const result = await this.prisma.salaryTerms.updateMany({
      where: { id, validTo: null },
      data: { validTo },
    });
    if (result.count === 0)
      throw new BadRequestException('Salary terms are closed');
    return this.get(id);
  }

  private async findNotStarted(db: Prisma.TransactionClient, id: string) {
    const row = await db.salaryTerms.findUnique({
      where: { id },
      select: { teacherId: true, validFrom: true, validTo: true },
    });
    if (!row) throw new NotFoundException('Salary terms not found');
    if (row.validFrom.getTime() <= kyivToday().getTime())
      throw new BadRequestException('Started salary terms cannot be changed');
    return row;
  }

  private validateAllowances(allowances: SalaryAllowanceDto[]) {
    return allowances.map((allowance) => ({
      name: allowance.name.trim(),
      amount: positive(allowance.amount),
    }));
  }

  private async validateRates(rates: SalaryRateDto[]) {
    const keys = new Set<string>();
    const subjectIds = new Set<string>();
    const groupIds = new Set<string>();
    for (const rate of rates) {
      if (rate.subjectId && rate.lessonType !== PayrollLessonType.school_lesson)
        throw new BadRequestException('Subject rate must be a school lesson');
      if (
        rate.classGroupId &&
        rate.lessonType === PayrollLessonType.school_lesson
      )
        throw new BadRequestException('Rate does not match class group format');
      const key = `${rate.lessonType}:${rate.subjectId ?? ''}:${rate.classGroupId ?? ''}`;
      if (keys.has(key)) throw new BadRequestException('Duplicate salary rate');
      keys.add(key);
      if (rate.subjectId) subjectIds.add(rate.subjectId);
      if (rate.classGroupId) groupIds.add(rate.classGroupId);
    }
    const [subjects, groups] = await Promise.all([
      this.prisma.subject.findMany({
        where: { id: { in: [...subjectIds] } },
        select: { id: true },
      }),
      this.prisma.classGroup.findMany({
        where: { id: { in: [...groupIds] } },
        select: { id: true, format: true },
      }),
    ]);
    if (subjects.length !== subjectIds.size)
      throw new NotFoundException('Subject not found');
    const formats = new Map(groups.map((group) => [group.id, group.format]));
    for (const rate of rates) {
      if (!rate.classGroupId) continue;
      const format = formats.get(rate.classGroupId);
      if (!format) throw new NotFoundException('Class group not found');
      const expected =
        format === ClassFormat.individual
          ? PayrollLessonType.individual
          : PayrollLessonType.class_group;
      if (rate.lessonType !== expected)
        throw new BadRequestException('Rate does not match class group format');
    }
    return rates.map((rate) => ({
      lessonType: rate.lessonType,
      subjectId: rate.subjectId ?? null,
      classGroupId: rate.classGroupId ?? null,
      amount: new Prisma.Decimal(rate.amount),
    }));
  }
}

      ...classes.map((lesson) => ({
        sourceType: PayrollSourceType.class_lesson,
        sourceId: lesson.id,
        date: lesson.date,
        lessonType:
          lesson.classGroup.format === ClassFormat.individual
            ? PayrollLessonType.individual
            : PayrollLessonType.class_group,
        subjectId: null,
        classGroupId: lesson.classGroupId,
        title: lesson.classGroup.name,
      })),
    ];
  }

  async compute(
    db: Db,
    teacherId: string,
    month: Date,
    workedDays: number | null,
    adjustments: AdjustmentLike[],
  ): Promise<ComputedSheet> {
    const to = lastDayOfMonth(month);
    const [terms, entries, lessons, approved] = await Promise.all([
      db.salaryTerms.findMany({
        where: { teacherId },
        include: { rates: true, allowances: true },
        orderBy: { validFrom: 'asc' },
      }),
      this.calendar.entriesBetween(month, to),
      this.lessons(db, teacherId, month, to),
      db.payrollSheet.findMany({
        where: {
          teacherId,
          status: PayrollStatus.approved,
          month: { lt: month },
        },
        select: { month: true },
        orderBy: { month: 'asc' },
      }),
    ]);
    let past: LessonSource[] = [];
    let booked: BookedLine[] = [];
    if (approved.length > 0) {
      const months = new Set(approved.map((sheet) => sheet.month.getTime()));
      const [before, lines] = await Promise.all([
        this.lessons(db, teacherId, approved[0].month, addDays(month, -1)),
        db.payrollLine.findMany({
          where: {
            sheet: { teacherId, status: PayrollStatus.approved },
            type: { in: [PayrollLineType.lesson, PayrollLineType.correction] },
            date: { lt: month },
          },
          select: {
            sourceType: true,
            sourceId: true,
            lessonType: true,
            date: true,
            title: true,
            amount: true,
          },
          orderBy: [{ date: 'asc' }, { id: 'asc' }],
        }),
      ]);
      past = before.filter((lesson) =>
        months.has(firstDayOfMonth(lesson.date).getTime()),
      );
      booked = lines.flatMap((line) =>
        line.sourceType && line.sourceId && line.lessonType && line.date
          ? [
              {
                sourceType: line.sourceType,
                sourceId: line.sourceId,
                lessonType: line.lessonType,
                date: line.date,
                title: line.title,
                amount: line.amount,
              },
            ]
          : [],
      );
    }
    return computeSheet({
      month,
      terms,
      entries,
      workedDays,
      lessons,
      past,
      booked,
      adjustments,
    });
  }

  async detail(db: Db, teacher: TeacherRef, month: Date, staff: boolean) {
    const sheet = await db.payrollSheet.findUnique({
      where: { teacherId_month: { teacherId: teacher.id, month } },
      include: sheetInclude,
    });
    let days: SheetDays;
    let lines: ComputedLine[];
    let warnings: { noRate: number; lessonsWithoutTerms: number };
    let lessons: number;
    if (sheet?.status === PayrollStatus.approved) {
      lines = sheet.lines.map(storedLine);
      days =
        (sheet.approvedDays as unknown as SheetDays | null) ?? EMPTY_DAYS;
      warnings = {
        noRate: lines.filter((line) => line.meta?.noRate).length,
        lessonsWithoutTerms: 0,
      };
      lessons = lines.filter((line) => line.type === PayrollLineType.lesson)
        .length;
    } else {
      const computed = await this.compute(
        db,
        teacher.id,
        month,
        sheet?.workedDays ?? null,
        sheet?.adjustments ?? [],
      );
      lines = computed.lines;
      days = computed.days;
      warnings = computed.warnings;
      lessons = computed.lessons;
    }
    const activePayments = (sheet?.payments ?? []).filter(
      (p) => p.voidedAt === null,
    );
    const accrued = sumAmounts(lines);
    const paid = sumAmounts(activePayments);
    return {
      teacher,
      month: monthKey(month),
      status: sheet?.status ?? PayrollStatus.draft,
      approvedAt: sheet?.approvedAt?.toISOString() ?? null,
      reopenReason: sheet?.reopenReason ?? null,
      days,
      lines: lines.map(lineView),
      adjustments: (sheet?.adjustments ?? [])
        .filter((a) => staff || a.voidedAt === null)
        .map((a) => ({
          id: a.id,
          kind: a.kind,
          amount: money(a.amount),
          reason: a.reason,
          createdAt: a.createdAt.toISOString(),
          ...(staff ? { createdBy: a.createdBy } : {}),
          voidedAt: a.voidedAt?.toISOString() ?? null,
          voidReason: a.voidReason,
        })),
      payments: (staff ? (sheet?.payments ?? []) : activePayments).map(
        (p) => ({
          id: p.id,
          date: isoDay(p.date),
          amount: money(p.amount),
          method: p.method,
          comment: p.comment,
          createdAt: p.createdAt.toISOString(),
          ...(staff ? { createdBy: p.createdBy } : {}),
          voidedAt: p.voidedAt?.toISOString() ?? null,
          voidReason: p.voidReason,
        }),
      ),
      warnings,
      lessons,
      totals: {
        accrued: money(accrued),
        paid: money(paid),
        balance: money(accrued.sub(paid)),
      },
    };
  }
}

import { BadRequestException, Injectable } from '@nestjs/common';
import { Prisma, Role } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import type { AuthenticatedUser } from '../common/types/authenticated-user.type';
import { parseMonth } from '../attendance/register-calc';
import { firstDayOfMonth, kyivToday, lastDayOfMonth } from '../tariffs/dates';
import { PayrollBuilder, toSummary } from './payroll-builder.service';

const ZERO = new Prisma.Decimal(0);

@Injectable()
export class PayrollService {
  constructor(
    private prisma: PrismaService,
    private builder: PayrollBuilder,
  ) {}

  async list(monthValue: string) {
    const { from, to } = parseMonth(monthValue);
    const teachers = await this.prisma.user.findMany({
      where: {
        role: Role.teacher,
        OR: [
          {
            salaryTerms: {
              some: {
                validFrom: { lte: to },
                OR: [{ validTo: null }, { validTo: { gte: from } }],
              },
            },
          },
          { payrollSheets: { some: { month: from } } },
        ],
      },
      select: { id: true, firstName: true, lastName: true },
      orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
    });
    const rows: ReturnType<typeof toSummary>[] = [];
    for (const teacher of teachers) {
      rows.push(
        toSummary(await this.builder.detail(this.prisma, teacher, from, true)),
      );
    }
    const total = (key: 'accrued' | 'paid' | 'balance') =>
      rows
        .reduce((sum, row) => sum.add(new Prisma.Decimal(row[key])), ZERO)
        .toFixed(2);
    return {
      month: monthValue,
      rows,
      totals: {
        accrued: total('accrued'),
        paid: total('paid'),
        balance: total('balance'),
      },
    };
  }

  async detail(teacherId: string, monthValue: string) {
    const { from } = parseMonth(monthValue);
    const teacher = await this.builder.teacher(this.prisma, teacherId);
    return this.builder.detail(this.prisma, teacher, from, true);
  }

  async myYear(user: AuthenticatedUser, yearValue: string) {
    if (!/^\d{4}$/.test(yearValue ?? ''))
      throw new BadRequestException('Invalid year');
    const year = Number(yearValue);
    const teacher = await this.builder.teacher(this.prisma, user.id);
    const from = new Date(Date.UTC(year, 0, 1));
    const to = new Date(Date.UTC(year, 11, 31));
    const current = firstDayOfMonth(kyivToday());
    const [terms, sheets] = await Promise.all([
      this.prisma.salaryTerms.findMany({
        where: {
          teacherId: teacher.id,
          validFrom: { lte: to },
          OR: [{ validTo: null }, { validTo: { gte: from } }],
        },
        select: { validFrom: true, validTo: true },
      }),
      this.prisma.payrollSheet.findMany({
        where: { teacherId: teacher.id, month: { gte: from, lte: to } },
        select: { month: true },
      }),
    ]);
    const months: {
      month: string;
      status: string;
      accrued: string;
      paid: string;
      balance: string;
    }[] = [];
    for (let index = 0; index < 12; index++) {
      const start = new Date(Date.UTC(year, index, 1));
      if (start.getTime() > current.getTime()) break;
      const end = lastDayOfMonth(start);
      const relevant =
        sheets.some((s) => s.month.getTime() === start.getTime()) ||
        terms.some(
          (t) =>
            t.validFrom.getTime() <= end.getTime() &&
            (t.validTo === null || t.validTo.getTime() >= start.getTime()),
        );
      if (!relevant) continue;
      const detail = await this.builder.detail(
        this.prisma,
        teacher,
        start,
        false,
      );
      months.push({
        month: detail.month,
        status: detail.status,
        accrued: detail.totals.accrued,
        paid: detail.totals.paid,
        balance: detail.totals.balance,
      });
    }
    return { year, months };
  }

  async myMonth(user: AuthenticatedUser, monthValue: string) {
    const { from } = parseMonth(monthValue);
    const teacher = await this.builder.teacher(this.prisma, user.id);
    return this.builder.detail(this.prisma, teacher, from, false);
  }
}

  async reopen(
    teacherId: string,
    monthValue: string,
    reason: string,
    user: AuthenticatedUser,
  ) {
    const { teacher, month } = await this.target(teacherId, monthValue);
    await this.prisma.$transaction(async (tx) => {
      await this.lockApproved(tx, teacher.id);
      const sheet = await tx.payrollSheet.findUnique({
        where: { teacherId_month: { teacherId: teacher.id, month } },
      });
      if (
        !sheet ||
        sheet.status !== PayrollStatus.approved ||
        !sheet.approvedAt
      )
        throw new BadRequestException('Payroll is not approved');
      const later = await tx.payrollPayment.count({
        where: {
          sheetId: sheet.id,
          voidedAt: null,
          createdAt: { gt: sheet.approvedAt },
        },
      });
      if (later > 0)
        throw new BadRequestException('Payroll has payments after approval');
      const corrections = await tx.payrollLine.count({
        where: {
          type: PayrollLineType.correction,
          refMonth: month,
          sheet: { teacherId: teacher.id, status: PayrollStatus.approved },
        },
      });
      if (corrections > 0)
        throw new BadRequestException('Payroll has later corrections');
      await tx.payrollLine.deleteMany({ where: { sheetId: sheet.id } });
      await tx.payrollSheet.update({
        where: { id: sheet.id },
        data: {
          status: PayrollStatus.draft,
          approvedAt: null,
          approvedById: null,
          approvedDays: Prisma.JsonNull,
          reopenedAt: new Date(),
          reopenedById: user.id,
          reopenReason: reason.trim(),
        },
      });
    });
    return this.builder.detail(this.prisma, teacher, month, true);
  }

  private async target(teacherId: string, monthValue: string) {
    const { from } = parseMonth(monthValue);
    const teacher = await this.builder.teacher(this.prisma, teacherId);
    return { teacher, month: from };
  }

  private async sheetDetail(sheet: { teacherId: string; month: Date }) {
    const teacher = await this.builder.teacher(this.prisma, sheet.teacherId);
    return this.builder.detail(this.prisma, teacher, sheet.month, true);
  }

  private async lockSheet(tx: Tx, teacherId: string, month: Date) {
    await tx.payrollSheet.createMany({
      data: [{ teacherId, month }],
      skipDuplicates: true,
    });
    const { id } = await tx.payrollSheet.findUniqueOrThrow({
      where: { teacherId_month: { teacherId, month } },
      select: { id: true },
    });
    return this.lockSheetById(tx, id);
  }

  private async lockSheetById(tx: Tx, id: string) {
    await tx.$queryRaw`SELECT id FROM "PayrollSheet" WHERE id = ${id} FOR UPDATE`;
    return tx.payrollSheet.findUniqueOrThrow({
      where: { id },
      include: {
        adjustments: { orderBy: [{ createdAt: 'asc' }, { id: 'asc' }] },
      },
    });
  }

  private async lockApproved(tx: Tx, teacherId: string) {
    await tx.$queryRaw`SELECT id FROM "PayrollSheet" WHERE "teacherId" = ${teacherId} AND status = 'approved' ORDER BY id FOR UPDATE`;
  }

### Domain: payroll

Spec: `docs/superpowers/specs/2026-10-13-payroll-design.md`. Code in `src/payroll/`:

- `SalaryTerms` versions per teacher (`validFrom`/`validTo`, never overlapping; a new version closes the open one; started versions are never edited, only closed; not started ones can be edited or deleted, deleting reopens the previous one) with `SalaryRate` (per lesson type, optional subject or class group; the most specific wins) and `SalaryAllowance` (monthly, prorated like the rate).

- `PayrollSheet` per teacher and month is created lazily. A `draft` is computed live by `PayrollBuilder`; `approve` snapshots `PayrollLine`s and `approvedDays`, and an approved sheet never changes. Lessons added or removed in an approved month become `correction` lines in later drafts (booked = `lesson` + `correction` lines of approved sheets for that lesson, dated before the draft month).
- Mutations lock the sheet row (`createMany skipDuplicates` + `FOR UPDATE`); `approve` then locks all approved sheets of the teacher (id-ordered); `reopen` locks that set first. Reopen is refused after a payment created after approval or a later correction for that month.
- `PayrollAdjustment` (bonus/deduction, drafts only) and `PayrollPayment` (cash/card, advances and overpayment allowed) are voided, never edited. Taxes are not tracked.
- Teachers read only their own payroll via `/me/payroll` (no staff names, no voided records). Users with salary terms or payroll records cannot be deleted; a teacher with salary terms cannot change role (400 `User has salary terms`).

const withRow = (list: PayrollList, detail: PayrollDetail): PayrollList => {
  const rows: PayrollRow[] = list.rows.map((row) =>
    row.teacher.id === detail.teacher.id
      ? {
          ...row,
          status: detail.status,
          accrued: detail.totals.accrued,
          paid: detail.totals.paid,
          balance: detail.totals.balance,
          lessons: detail.lessons,
          warnings: detail.warnings,
        }
      : row,
  );
  const total = (key: 'accrued' | 'paid' | 'balance') =>
    fromKopecks(rows.reduce((sum, row) => sum + toKopecks(row[key]), 0));
  return {
    ...list,
    rows,
    totals: { accrued: total('accrued'), paid: total('paid'), balance: total('balance') },
  };
};

            name: `${direction.name}: ${group.name}`,
            format: group.format,
          })),
        ),
      });
    } catch (error) {
      set({ refsFailed: true });
      throw error;
    }
  },

  month: localToday().slice(0, 7),
  list: null,
  listFailed: false,
  setMonth: (month) => {
    set({ month, list: null, listFailed: false, detailTeacherId: null, detail: null });
    get().loadList().catch(() => undefined);
  },
  loadList: async () => {
    const month = get().month;
    set({ listFailed: false });
    try {
      const { data } = await payrollApi.getList(month);
      if (get().month === month) set({ list: data });
    } catch (error) {
      if (get().month === month) set({ listFailed: true });
      throw error;
    }
  },

  detailTeacherId: null,
  detail: null,
  detailFailed: false,
  isSaving: false,
  openDetail: (teacherId) => {
    set({ detailTeacherId: teacherId, detail: null, detailFailed: false });
    get().loadDetail().catch(() => undefined);
  },
  closeDetail: () => set({ detailTeacherId: null, detail: null, detailFailed: false }),
  loadDetail: async () => {
    const { detailTeacherId: teacherId, month } = get();
    if (!teacherId) return;
    const current = () => get().detailTeacherId === teacherId && get().month === month;
    set({ detailFailed: false });
    try {
      const { data } = await payrollApi.getDetail(teacherId, month);
      if (current()) set({ detail: data });
    } catch (error) {
      if (current()) set({ detailFailed: true });
      throw error;
    }
  },
  runDetail: async (action) => {
    const { detailTeacherId: teacherId, month } = get();
    if (!teacherId) return;
    set({ isSaving: true });
    try {
      const { data } = await action(teacherId, month);
      const { list } = get();
      if (get().detailTeacherId === teacherId && get().month === month) set({ detail: data });
      if (list && list.month === month) set({ list: withRow(list, data) });
    } finally {
      set({ isSaving: false });
    }
  },

  termsTeacherId: '',
  terms: [],
  termsLoading: false,
  termsFailed: false,
  setTermsTeacher: (teacherId) => {
    set({ termsTeacherId: teacherId, terms: [], termsFailed: false });
    if (teacherId) get().loadTerms().catch(() => undefined);
  },
  loadTerms: async () => {
    const teacherId = get().termsTeacherId;
    if (!teacherId) return;
    set({ termsLoading: true, termsFailed: false });
    try {
      const { data } = await payrollApi.getTerms(teacherId);
      if (get().termsTeacherId === teacherId) set({ terms: data });
    } catch (error) {
      if (get().termsTeacherId === teacherId) set({ termsFailed: true });
      throw error;
    } finally {
      if (get().termsTeacherId === teacherId) set({ termsLoading: false });
    }
  },
  runTerms: async (action) => {
    set({ isSaving: true });
    try {
      await action();
      await get().loadTerms();
    } finally {
      set({ isSaving: false });
    }
  },
}));

  }
  const voidTarget = modal && (modal.kind === 'voidAdjustment' || modal.kind === 'voidPayment') ? modal : null

  return (
    <Dialog open fullWidth maxWidth="md" onClose={closeDetail}>
      <DialogTitle>
        {detail ? `${detail.teacher.lastName} ${detail.teacher.firstName}` : 'Відомість'}
        {detail && (
          <Typography component="span" color="text.secondary" sx={{ ml: 1 }}>
            {monthLabel(detail.month)}
          </Typography>
        )}
      </DialogTitle>
      <DialogContent dividers>
        {detailFailed ? (
          <Alert
            severity="error"
            action={<Button onClick={() => loadDetail().catch(() => undefined)}>Спробувати ще</Button>}
          >
            Не вдалося завантажити дані.
          </Alert>
        ) : !detail ? (
          <Typography color="text.secondary">Завантаження…</Typography>
        ) : (
          <Stack spacing={2}>
            <Stack direction="row" spacing={1} alignItems="center">
              <Chip
                size="small"
                label={payrollStatusLabels[detail.status]}
                color={detail.status === 'approved' ? 'success' : 'default'}
              />
              {draft && detail.reopenReason && (
                <Typography variant="body2" color="text.secondary">
                  Відкрито повторно: {detail.reopenReason}
                </Typography>
              )}
            </Stack>
            {draft && (
              <WorkedDaysForm key={`${detail.days.workedDays}-${String(detail.days.override)}`} detail={detail} />
            )}
            <PayrollDetailView
              detail={detail}
              onVoidAdjustment={draft ? (id) => setModal({ kind: 'voidAdjustment', id }) : undefined}
              onVoidPayment={(id) => setModal({ kind: 'voidPayment', id })}
            />
          </Stack>
        )}
      </DialogContent>
      <DialogActions sx={{ flexWrap: 'wrap', gap: 1 }}>
        {detail && draft && (
          <Button disabled={isSaving} onClick={() => setModal({ kind: 'adjustment' })}>
            Доплата / утримання
          </Button>
        )}
        {detail && (
          <Button disabled={isSaving} onClick={() => setModal({ kind: 'payment' })}>
            Виплата
          </Button>
        )}
        {detail && draft && (
          <Button variant="contained" disabled={isSaving} onClick={approve}>
            Затвердити
          </Button>
        )}
        {detail && !draft && (
          <Button color="warning" disabled={isSaving} onClick={() => setModal({ kind: 'reopen' })}>
            Відкрити повторно
          </Button>
        )}
        <Button onClick={closeDetail}>Закрити</Button>
      </DialogActions>

      {modal?.kind === 'adjustment' && <AdjustmentDialog onClose={closeModal} />}
      {modal?.kind === 'payment' && <PaymentDialog onClose={closeModal} />}
      {modal?.kind === 'reopen' && (
        <ReasonDialog
          title="Відкрити відомість повторно"
          onClose={closeModal}
          onSubmit={(reason) => runDetail((teacherId, month) => payrollApi.reopen(teacherId, month, reason))}
        />
      )}
      {voidTarget && (
        <ReasonDialog
          title={voidTarget.kind === 'voidAdjustment' ? 'Анулювати запис' : 'Анулювати виплату'}
          onClose={closeModal}
          onSubmit={(reason) =>
            runDetail(() =>
              voidTarget.kind === 'voidAdjustment'
                ? payrollApi.voidAdjustment(voidTarget.id, reason)
                : payrollApi.voidPayment(voidTarget.id, reason),
            )
          }
        />
      )}
    </Dialog>
  )

import { create } from 'zustand';
import { localToday, payrollApi, type MyPayrollYear, type PayrollDetail } from '../../../shared/api';

interface MyPayrollState {
  year: number;
  data: MyPayrollYear | null;
  failed: boolean;
  setYear: (year: number) => void;
  load: () => Promise<void>;
  month: string | null;
  detail: PayrollDetail | null;
  detailFailed: boolean;
  openMonth: (month: string) => void;
  closeMonth: () => void;
  loadMonth: () => Promise<void>;
}

export const useMyPayrollStore = create<MyPayrollState>((set, get) => ({
  year: Number(localToday().slice(0, 4)),
  data: null,
  failed: false,
  setYear: (year) => {
    set({ year, data: null, failed: false });
    get().load().catch(() => undefined);
  },
  load: async () => {
    const year = get().year;
    set({ failed: false });
    try {
      const { data } = await payrollApi.getMyYear(year);
      if (get().year === year) set({ data });
    } catch (error) {
      if (get().year === year) set({ failed: true });
      throw error;
    }
  },
  month: null,
  detail: null,
  detailFailed: false,
  openMonth: (month) => {
    set({ month, detail: null, detailFailed: false });
    get().loadMonth().catch(() => undefined);
  },
  closeMonth: () => set({ month: null, detail: null, detailFailed: false }),
  loadMonth: async () => {
    const month = get().month;
    if (!month) return;
    set({ detailFailed: false });
    try {
      const { data } = await payrollApi.getMyMonth(month);
      if (get().month === month) set({ detail: data });
    } catch (error) {
      if (get().month === month) set({ detailFailed: true });
      throw error;
    }
  },
}));
