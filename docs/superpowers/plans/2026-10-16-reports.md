```ts
import {
  BadRequestException,
  ForbiddenException,
  NotFoundException,
} from '@nestjs/common';
import { GroupKind, Prisma } from '@prisma/client';
import { scopeKind } from '../access/access-calc';
import { OUTSIDE_SCOPE } from '../access/scope';
import { parseDay } from '../classes/access';
import type { AuthenticatedUser } from '../common/types/authenticated-user.type';
import { dayCount } from '../tariffs/dates';

export const MAX_PERIOD_DAYS = 366;

export interface ReportFilter {
  kind: GroupKind | null;
  groupId: string | null;
}

export function reportPeriod(
  from?: string,
  to?: string,
): { from: Date; to: Date } {
  const start = parseDay(from ?? '');
  const end = parseDay(to ?? '');
  if (end.getTime() < start.getTime())
    throw new BadRequestException('Invalid period');
  if (dayCount(start, end) > MAX_PERIOD_DAYS)
    throw new BadRequestException('Period too long');
  return { from: start, to: end };
}

export async function resolveFilter(
  db: {
    group: {
      findUnique(args: {
        where: { id: string };
        select: { id: true; kind: true };
      }): Promise<{ id: string; kind: GroupKind } | null>;
    };
  },
  user: AuthenticatedUser,
  query: { kind?: GroupKind; groupId?: string },
): Promise<ReportFilter> {
  const scope = scopeKind(user);
  if (scope && query.kind && query.kind !== scope)
    throw new ForbiddenException(OUTSIDE_SCOPE);
  if (query.groupId) {
    const group = await db.group.findUnique({
      where: { id: query.groupId },
      select: { id: true, kind: true },
    });
    if (!group) throw new NotFoundException('Group not found');
    if (scope && group.kind !== scope)
      throw new ForbiddenException(OUTSIDE_SCOPE);
  }
  return { kind: query.kind ?? scope, groupId: query.groupId ?? null };
}

export const childFilterWhere = (
  filter: ReportFilter,
  childId?: string,
): Prisma.ChildWhereInput => ({
  ...(filter.kind ? { group: { kind: filter.kind } } : {}),
  ...(filter.groupId ? { groupId: filter.groupId } : {}),
  ...(childId ? { id: childId } : {}),
});
```

```ts
import ExcelJS from 'exceljs';
import { uaDay } from '../tariffs/dates';
import type { Cell, ColumnType, Report, ReportColumn, ReportRow } from './report.types';

const FORMATS: Partial<Record<ColumnType, string>> = {
  money: '0.00',
  date: 'dd.mm.yyyy',
  percent: '0%',
};

const SHEET_NAME_MAX = 31;

const cellValue = (type: ColumnType, value: Cell): ExcelJS.CellValue => {
  if (value === null || value === '') return null;
  if (type === 'money') return Number(Number(value).toFixed(2));
  if (type === 'date') return new Date(`${String(value)}T00:00:00.000Z`);
  return value;
};

const title = (report: Report) =>
  report.period
    ? `${report.title} · ${uaDay(new Date(`${report.period.from}T00:00:00.000Z`))} – ${uaDay(new Date(`${report.period.to}T00:00:00.000Z`))}`
    : report.title;

function writeSheet(
  book: ExcelJS.Workbook,
  name: string,
  heading: string,
  columns: ReportColumn[],
  rows: ReportRow[],
  totals: ReportRow | null,
) {
  const sheet = book.addWorksheet(name.slice(0, SHEET_NAME_MAX), {
    views: [{ state: 'frozen', ySplit: 3 }],
  });
  sheet.getCell(1, 1).value = heading;
  sheet.getCell(1, 1).font = { bold: true, size: 14 };
  const header = sheet.getRow(3);
  columns.forEach((column, index) => {
    const cell = header.getCell(index + 1);
    cell.value = column.label;
    cell.font = { bold: true };
  });
  const lines = totals ? [...rows, totals] : rows;
  lines.forEach((row, rowIndex) => {
    const line = sheet.getRow(4 + rowIndex);
    columns.forEach((column, index) => {
      const cell = line.getCell(index + 1);
      cell.value = cellValue(column.type, row[column.key] ?? null);
      const format = FORMATS[column.type];
      if (format) cell.numFmt = format;
      if (totals && rowIndex === rows.length) cell.font = { bold: true };
    });
  });
  columns.forEach((column, index) => {
    const width = Math.max(
      column.label.length,
      ...lines.map((row) => String(row[column.key] ?? '').length),
    );
    sheet.getColumn(index + 1).width = Math.min(Math.max(width + 2, 10), 60);
  });
}

export async function renderXlsx(report: Report): Promise<Buffer> {
  const book = new ExcelJS.Workbook();
  writeSheet(book, report.title, title(report), report.columns, report.rows, report.totals);
  for (const extra of report.sheets)
    writeSheet(book, extra.name, `${title(report)} · ${extra.name}`, extra.columns, extra.rows, extra.totals);
  return Buffer.from(await book.xlsx.writeBuffer());
}
```

```ts
export type ReportKey = 'payments' | 'charges' | 'debts' | 'attendance' | 'lessons' | 'meals' | 'payroll';
export type ColumnType = 'text' | 'money' | 'number' | 'date' | 'percent';
export interface ReportColumn { key: string; label: string; type: ColumnType }
export type Cell = string | number | null;
export type ReportRow = Record<string, Cell>;
export interface ReportSheet { name: string; columns: ReportColumn[]; rows: ReportRow[]; totals: ReportRow | null }
export interface Report { title: string; fileName: string; period: { from: string; to: string } | null; columns: ReportColumn[]; rows: ReportRow[]; totals: ReportRow | null; sheets: ReportSheet[] }
export interface ReportQuery { from?: string; to?: string; kind?: 'kindergarten' | 'school_class'; groupId?: string; childId?: string; teacherId?: string }
export const reportsApi: {
  get(key: ReportKey, query: ReportQuery): Promise<Report>;
  download(key: ReportKey, query: ReportQuery, fileName: string): Promise<void>;
};
export const reportDefinitions: { key: ReportKey; label: string; areas: PermissionArea[]; period: boolean; filters: ('kind' | 'group' | 'child' | 'teacher')[] }[];
export const formatCell: (type: ColumnType, value: Cell) => string;
```

`download` uses `axiosInstance.get(url, { params: { ...query, format: 'xlsx' }, responseType: 'blob' })`, creates an object URL and clicks a temporary `<a download={fileName}>`. The file name is built on the client (`${fileName}_${from}_${to}.xlsx`, or `_${today}` for debts) because the backend's `Content-Disposition` header is not exposed cross-origin and CORS must not change.

`reportDefinitions` (verbatim labels): payments «Оплати» [finance] period, filters kind/group/child; charges «Нарахування» [finance] same; debts «Борги» [finance] no period, kind/group/child; attendance «Відвідування» [register] period, kind/group/child; lessons «Уроки» [timetable, classes] period, kind/group/teacher; meals «Харчування» [meals] period, kind/group; payroll «Зарплати» [payroll] period, teacher.

`formatCell`: money → `Number(v).toLocaleString('uk-UA', { minimumFractionDigits: 2, maximumFractionDigits: 2 })`; date → `DD.MM.YYYY` from the ISO day (no time zone shift); percent → `Math.round(v * 100) + '%'`; null → `''`.
