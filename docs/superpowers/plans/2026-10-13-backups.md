> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The backend dumps the Postgres database every night at 02:00 Kyiv (and on demand) into a cloud-synced folder, keeps 30 daily + 12 monthly copies, shows the state in the admin panel, and a documented command restores a copy.

**Architecture:** New backend module `src/backups/` with two pure files (`backup-retention.ts`: file names, Kyiv stamps, retention; `backup-command.ts`: `DATABASE_URL` parsing, `pg_dump`/`pg_restore` command specs, secret masking), two thin I/O classes (`ProcessRunner` spawning processes, `BackupStorage` over `fs/promises`), `BackupsService` (create/status/runs, one run at a time), a cron `BackupsScheduler`, and a controller. `scripts/backup-restore.ts` restores a copy from the command line. The frontend adds a «Резервні копії» page and a dashboard banner.

```ts
import { BackupKind } from '@prisma/client';
import { addDays, isoDay } from '../tariffs/dates';

export interface BackupFileInfo {
  name: string;
  kind: BackupKind;
  stamp: string;
}

const KEEP_DAILY = 30;
const KEEP_MONTHS = 12;
const MANUAL_DAYS = 30;

const SUFFIX: Record<BackupKind, string> = {
  [BackupKind.scheduled]: '',
  [BackupKind.manual]: '_manual',
  [BackupKind.pre_restore]: '_pre-restore',
};

const PATTERN =
  /^edu_db_(\d{4}-(?:0[1-9]|1[0-2])-(?:0[1-9]|[12]\d|3[01])_(?:[01]\d|2[0-3])[0-5]\d[0-5]\d)(_manual|_pre-restore)?\.dump$/;

export function kyivStamp(now: Date): string {
  const parts = Object.fromEntries(
    new Intl.DateTimeFormat('en-GB', {
      timeZone: 'Europe/Kyiv',
      year: 'numeric',
      month: '2-digit',
      day: '2-digit',
      hour: '2-digit',
      minute: '2-digit',
      second: '2-digit',
      hourCycle: 'h23',
    })
      .formatToParts(now)
      .map((part) => [part.type, part.value]),
  );
  return `${parts.year}-${parts.month}-${parts.day}_${parts.hour}${parts.minute}${parts.second}`;
}

export const backupFileName = (kind: BackupKind, now: Date): string =>
  `edu_db_${kyivStamp(now)}${SUFFIX[kind]}.dump`;

export function parseBackupName(name: string): BackupFileInfo | null {
  const match = PATTERN.exec(name);
  if (!match) return null;
  const kind =
    match[2] === '_manual'
      ? BackupKind.manual
      : match[2] === '_pre-restore'
        ? BackupKind.pre_restore
        : BackupKind.scheduled;
  return { name, kind, stamp: match[1] };
}

export const takenAtOf = (stamp: string): string =>
  `${stamp.slice(0, 10)}T${stamp.slice(11, 13)}:${stamp.slice(13, 15)}:${stamp.slice(15, 17)}`;

const newestFirst = (a: BackupFileInfo, b: BackupFileInfo) =>
  b.stamp.localeCompare(a.stamp);

export function filesToDelete(
  files: BackupFileInfo[],
  today: Date,
): string[] {
  const keep = new Set<string>();
  const scheduled = files
    .filter((f) => f.kind === BackupKind.scheduled)
    .sort(newestFirst);
  for (const file of scheduled.slice(0, KEEP_DAILY)) keep.add(file.name);
  const months = new Map<string, BackupFileInfo>();
  for (const file of scheduled) {
    const month = file.stamp.slice(0, 7);
    if (!months.has(month)) months.set(month, file);
  }
  [...months.keys()]
    .sort((a, b) => b.localeCompare(a))
    .slice(0, KEEP_MONTHS)
    .forEach((month) => keep.add(months.get(month)!.name));
  const manualLimit = isoDay(addDays(today, -MANUAL_DAYS));
  const deleted = files.filter((file) => {
    if (file.kind === BackupKind.pre_restore) return false;
    if (file.kind === BackupKind.manual)
      return file.stamp.slice(0, 10) < manualLimit;
    return !keep.has(file.name);
  });
  return deleted.map((f) => f.name).sort();
}
```

```ts
export interface DbConfig {
  user: string;
  password: string;
  host: string;
  port: string;
  database: string;
}

export interface CommandSpec {
  command: string;
  args: string[];
  env: Record<string, string>;
}

const MAX_ERROR = 2000;

export function parseDatabaseUrl(url: string): DbConfig {
  let parsed: URL;
  try {
    parsed = new URL(url);
  } catch {
    throw new Error('Invalid DATABASE_URL');
  }
  const database = decodeURIComponent(parsed.pathname.replace(/^\//, ''));
  if (!parsed.hostname || !database) throw new Error('Invalid DATABASE_URL');
  return {
    user: decodeURIComponent(parsed.username),
    password: decodeURIComponent(parsed.password),
    host: parsed.hostname,
    port: parsed.port || '5432',
    database,
  };
}

export function passwordOf(url: string | undefined): string {
  try {
    return parseDatabaseUrl(url ?? '').password;
  } catch {
    return '';
  }
}

const passwordEnv = (db: DbConfig) => ({ PGPASSWORD: db.password });

const connection = (db: DbConfig) => [
  '-h',
  db.host,
```

```ts
import { spawn } from 'node:child_process';
import { createReadStream, createWriteStream } from 'node:fs';
import type { CommandSpec } from './backup-command';

export interface RunOptions {
  stdoutFile?: string;
  stdinFile?: string;
  timeoutMs: number;
}

export interface RunResult {
  code: number | null;
  stderr: string;
  timedOut: boolean;
}

const MAX_STDERR = 10_000;

export class ProcessRunner {
  run(spec: CommandSpec, options: RunOptions): Promise<RunResult> {
    return new Promise((resolve) => {
      const child = spawn(spec.command, spec.args, {
        env: { ...process.env, ...spec.env },
        windowsHide: true,
        stdio: [
          options.stdinFile ? 'pipe' : 'ignore',
          options.stdoutFile ? 'pipe' : 'ignore',
          'pipe',
        ],
      });
      let stderr = '';
      let timedOut = false;
      let settled = false;
      let exitCode: number | null = null;
      let exited = false;
      let flushed = !options.stdoutFile;
      const finish = (code: number | null) => {
        if (settled) return;
        settled = true;
        clearTimeout(timer);
        resolve({ code, stderr, timedOut });
      };
      const maybeFinish = () => {
        if (exited && flushed) finish(exitCode);
      };
      const timer = setTimeout(() => {
        timedOut = true;
        child.kill();
      }, options.timeoutMs);
      child.stderr?.on('data', (chunk: Buffer) => {
        if (stderr.length < MAX_STDERR) stderr += chunk.toString();
      });
      child.on('error', (error) => {
        stderr += error.message;
        finish(null);
      });
      if (options.stdoutFile && child.stdout) {
        const out = createWriteStream(options.stdoutFile);
        out.on('finish', () => {
          flushed = true;
          maybeFinish();
        });
        out.on('error', (error) => {
          stderr += error.message;
          flushed = true;
          maybeFinish();
        });
        child.stdout.pipe(out);
      }
      if (options.stdinFile && child.stdin) {
        const input = createReadStream(options.stdinFile);
        input.on('error', (error) => {
          stderr += error.message;
          child.kill();
        });
        child.stdin.on('error', () => undefined);
        input.pipe(child.stdin);
      }
      child.on('close', (code) => {
        exitCode = code;
        exited = true;
        maybeFinish();
      });
    });
  }
}
```

```ts
import { join } from 'node:path';

export class BackupStorage {
  async ensureDir(dir: string): Promise<void> {
    await mkdir(dir, { recursive: true });
  }

  async list(dir: string): Promise<{ name: string; size: number }[]> {
    const names = await readdir(dir).catch(() => [] as string[]);
    const files: { name: string; size: number }[] = [];
    for (const name of names) {
      const info = await stat(join(dir, name)).catch(() => null);
      if (info?.isFile()) files.push({ name, size: info.size });
    }
    return files;
  }

  size(path: string): Promise<number> {
    return stat(path)
      .then((info) => info.size)
      .catch(() => 0);
  }

  rename(from: string, to: string): Promise<void> {
    return rename(from, to);
  }

  remove(path: string): Promise<void> {
    return rm(path, { force: true });
  }
}
```

```ts
    if (this.running) throw new ConflictException('Backup is already running');
    this.running = true;
    const databaseUrl = this.config.get<string>('DATABASE_URL');
    const secret = passwordOf(databaseUrl);
    try {
      const run = await this.prisma.backupRun.create({
        data: { kind, status: BackupStatus.running, createdById: userId },
      });
      try {
        const db = parseDatabaseUrl(databaseUrl ?? '');
        const name = backupFileName(kind, new Date());
        const target = join(dir, name);
        const part = `${target}.part`;
        await this.storage.ensureDir(dir);
        const { spec, stdoutToFile } = dumpCommand(db, container, part);
        const result = await this.runner.run(spec, {
          stdoutFile: stdoutToFile ? part : undefined,
          timeoutMs: TIMEOUT_MS,
        });
        const size = await this.storage.size(part);
        if (result.timedOut || result.code !== 0 || size === 0) {
          await this.storage.remove(part);
          const error = result.timedOut
            ? 'Timed out'
            : result.code !== 0
              ? maskSecret(
                  result.stderr.trim() || `Exit code ${result.code}`,
                  secret,
                )
              : 'Empty dump';
          return this.finish(run.id, { status: BackupStatus.failed, error });
        }
        await this.storage.rename(part, target);
        const view = await this.finish(run.id, {
          status: BackupStatus.succeeded,
          fileName: name,
          sizeBytes: BigInt(size),
        });
        await this.applyRetention(dir);
        return view;
      } catch (error) {
        return this.finish(run.id, {
          status: BackupStatus.failed,
          error: maskSecret(messageOf(error), secret),
        });
      }
    } finally {
      this.running = false;
    }
  }

  async status() {
    const { dir } = this.settings();
    const [lastSuccess, lastFailure, lastRun] = await Promise.all([
      this.prisma.backupRun.findFirst({
        where: { status: BackupStatus.succeeded },
        orderBy: { startedAt: 'desc' },
        include: runInclude,
      }),
      this.prisma.backupRun.findFirst({
        where: { status: BackupStatus.failed },
        orderBy: { startedAt: 'desc' },
        include: runInclude,
      }),
      this.prisma.backupRun.findFirst({
        where: { status: { not: BackupStatus.running } },
        orderBy: { startedAt: 'desc' },
        include: runInclude,
      }),
    ]);
    const files = dir
      ? (await this.storage.list(dir))
          .flatMap((file) => {
            const info = parseBackupName(file.name);
            return info
              ? [
                  {
                    name: file.name,
                    kind: info.kind,
                    sizeBytes: file.size,
                    takenAt: takenAtOf(info.stamp),
                    stamp: info.stamp,
                  },
                ]
              : [];
          })
          .sort((a, b) => b.stamp.localeCompare(a.stamp))
          .map(({ stamp: _stamp, ...file }) => file)
      : [];
    const successAt = lastSuccess?.finishedAt?.getTime();
    const stale =
      !!dir &&
      (successAt === undefined ||
        Date.now() - successAt > STALE_MS ||
        lastRun?.status === BackupStatus.failed);
    return {
      configured: !!dir,
      dir,
      running: this.running,
      lastSuccess: lastSuccess ? toRunView(lastSuccess) : null,
      lastFailure: lastFailure ? toRunView(lastFailure) : null,
      stale,
      files,
    };
  }

  async runs() {
    const rows = await this.prisma.backupRun.findMany({
      orderBy: { startedAt: 'desc' },
      take: 50,
      include: runInclude,
    });
    return rows.map(toRunView);
  }

  private async finish(
    id: string,
    data: Pick<
      Prisma.BackupRunUpdateInput,
      'status' | 'error' | 'fileName' | 'sizeBytes'
    >,
  ) {
    const run = await this.prisma.backupRun.update({
      where: { id },
      data: { ...data, finishedAt: new Date() },
      include: runInclude,
    });
    return toRunView(run);
  }

  private async applyRetention(dir: string) {
    try {
      const files = (await this.storage.list(dir)).flatMap((file) => {
        const info = parseBackupName(file.name);
        return info ? [info] : [];
      });
      for (const name of filesToDelete(files, kyivToday())) {
        await this.storage
          .remove(join(dir, name))
          .catch((error) =>
            this.logger.warn(`Could not delete ${name}: ${messageOf(error)}`),
          );
      }
    } catch (error) {
      this.logger.warn(`Retention failed: ${messageOf(error)}`);
    }
  }
}
```

```ts
import 'dotenv/config';
import { BackupKind } from '@prisma/client';
import { existsSync, rmSync } from 'node:fs';
import { connect } from 'node:net';
import { join, resolve } from 'node:path';
import { createInterface } from 'node:readline/promises';
import {
  dumpCommand,
  maskSecret,
  parseDatabaseUrl,
  restoreCommand,
} from '../src/backups/backup-command';
import { backupFileName } from '../src/backups/backup-retention';
import { ProcessRunner } from '../src/backups/process-runner';

const TIMEOUT_MS = 60 * 60 * 1000;

const fail = (message: string): never => {
  console.error(message);
  process.exit(1);
};

const portBusy = (port: number) =>
  new Promise<boolean>((done) => {
    const socket = connect({ port, host: '127.0.0.1' });
    socket.once('connect', () => {
      socket.destroy();
      done(true);
    });
    socket.once('error', () => done(false));
  });

async function main() {
  const args = process.argv.slice(2);
  const yes = args.includes('--yes');
  const target = args.find((arg) => arg !== '--yes');
  if (!target) fail('Usage: npm run backup:restore -- <file> [--yes]');
  const dir = process.env.BACKUP_DIR?.trim() ?? '';
  if (!dir) fail('BACKUP_DIR is not set: a safety copy is required before restore');
  const file = /[\\/]/.test(target!) ? resolve(target!) : join(dir, target!);
  if (!existsSync(file)) fail(`File not found: ${file}`);
  const port = Number(process.env.PORT ?? 3000);
  if (await portBusy(port)) fail('Stop the backend first');
  const db = parseDatabaseUrl(process.env.DATABASE_URL ?? '');
  const raw = process.env.BACKUP_PG_CONTAINER;
  const container = raw === undefined ? 'edu_postgres' : raw.trim() || null;
  if (!yes) {
    const prompt = createInterface({
      input: process.stdin,
      output: process.stdout,
    });
    const answer = await prompt.question(
      `This replaces all data in "${db.database}". Type the database name to continue: `,
    );
    prompt.close();
    if (answer.trim() !== db.database) fail('Cancelled');
  }
  const runner = new ProcessRunner();
  const safety = join(dir, backupFileName(BackupKind.pre_restore, new Date()));
  const dump = dumpCommand(db, container, safety);
  const dumped = await runner.run(dump.spec, {
    stdoutFile: dump.stdoutToFile ? safety : undefined,
    timeoutMs: TIMEOUT_MS,
  });
  if (dumped.timedOut || dumped.code !== 0) {
    rmSync(safety, { force: true });
    fail(
      `Safety copy failed, nothing restored: ${maskSecret(dumped.stderr, db.password)}`,
    );
  }
  console.log(`Safety copy: ${safety}`);
  const restore = restoreCommand(db, container, file);
  const restored = await runner.run(restore.spec, {
    stdinFile: restore.stdinFromFile ? file : undefined,
    timeoutMs: TIMEOUT_MS,
  });
  if (restored.timedOut || restored.code !== 0)
    fail(
      `Restore failed (safety copy kept: ${safety}):\n${maskSecret(restored.stderr, db.password)}`,
    );
  console.log(`Restored ${file} into ${db.database}`);
}

main().catch((error: unknown) =>
  fail(error instanceof Error ? error.message : String(error)),
);
```

```ts
import { create } from 'zustand';
import { backupsApi, type BackupRun, type BackupStatus } from '../../../shared/api';
import { useNotifyStore } from '../../../shared/model/notifyStore';

interface BackupsState {
  status: BackupStatus | null;
  runs: BackupRun[];
  failed: boolean;
  isCreating: boolean;
  load: () => Promise<void>;
  create: () => Promise<void>;
}

export const useBackupsStore = create<BackupsState>((set, get) => ({
  status: null,
  runs: [],
  failed: false,
  isCreating: false,
  load: async () => {
    set({ failed: false });
    try {
      const [status, runs] = await Promise.all([backupsApi.getStatus(), backupsApi.getRuns()]);
      set({ status: status.data, runs: runs.data });
    } catch (error) {
      set({ failed: true });
      throw error;
    }
  },
  create: async () => {
    set({ isCreating: true });
    try {
      const { data } = await backupsApi.create();
      if (data.status === 'succeeded') useNotifyStore.getState().push('Копію створено', 'success');
      else useNotifyStore.getState().push(`Копію не створено: ${data.error ?? 'помилка'}`);
      await get().load();
    } finally {
      set({ isCreating: false });
    }
  },
}));
```
