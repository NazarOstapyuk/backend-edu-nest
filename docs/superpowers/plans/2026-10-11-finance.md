# Finance Module (Part 1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Per-child finance ledger (charges, cash/bank payments, allocations, voiding) with balances by category, plus admin UI in the frontend.

**Architecture:** Three new Prisma tables (`Charge`, `Payment`, `Allocation`); balances are never stored, they are computed by pure functions (`finance-calc.ts`) and shaped into API views by pure functions (`finance-view.ts`). Write services (`ChargesService`, `PaymentsService` + `applyAllocations`) lock rows with `SELECT ... FOR UPDATE` inside `$transaction`. Every mutation endpoint returns the child's full `ChildFinance` view, so the frontend replaces its state with the response.

**Tech Stack:** NestJS 11, Prisma 7 (`prisma-client-js`, adapter-pg), class-validator, Jest. Frontend: React 19, MUI 9, Zustand 5, react-hook-form 7 + yup, axios.

**Spec:** `docs/superpowers/specs/2026-10-11-finance-design.md`

1. Two admins allocate the same payment at the same time → total allocated must still not exceed the payment (row locks taken before reading sums). Pinned in Task 4 (lock order test).
2. Amount `"0"` / `"0.00"` passes the regex → must be rejected with `Amount must be greater than 0`. Pinned in Tasks 3, 4, 5.

```ts
import { FinanceCategory, PaymentStatus, Prisma } from '@prisma/client';

export const AMOUNT_PATTERN = /^\d{1,8}(\.\d{1,2})?$/;

export type Money = Prisma.Decimal;
export type MoneyInput = string | number | Prisma.Decimal;
export type ChargeState = 'unpaid' | 'partial' | 'paid';

export interface AllocationLike {
  amount: Money;
  voidedAt: Date | null;
}

export interface ChargeLike {
  id: string;
  category: FinanceCategory;
  amount: Money;
  description: string;
  periodStart: Date | null;
  createdAt: Date;
  voidedAt: Date | null;
  allocations: AllocationLike[];
}

export interface PaymentLike {
  amount: Money;
  category: FinanceCategory | null;
  status: PaymentStatus;
  voidedAt: Date | null;
  allocations: AllocationLike[];
}

export interface CategorySummary {
  category: FinanceCategory;
  charged: string;
  paid: string;
  debt: string;
  prepayment: string;
}

export interface FinanceSummary {
  totals: { totalDue: string; unallocated: string };
  categories: CategorySummary[];
}

export interface SuggestionLine {
  chargeId: string;
  category: FinanceCategory;
  description: string;
  periodStart: Date | null;
  createdAt: Date;
  remaining: string;
  suggestedAmount: string;
}

export const money = (value: MoneyInput): Money => new Prisma.Decimal(value);

export const formatMoney = (value: MoneyInput): string => {
  const decimal = new Prisma.Decimal(value);
  return (decimal.isZero() ? new Prisma.Decimal(0) : decimal).toFixed(2);
};

export const sumMoney = (values: MoneyInput[]): Money =>
  values.reduce<Money>((acc, value) => acc.plus(value), new Prisma.Decimal(0));

export const activeAllocated = (allocations: AllocationLike[]): Money =>
  sumMoney(allocations.filter((a) => !a.voidedAt).map((a) => a.amount));

export const isCountedPayment = (payment: PaymentLike): boolean =>
  !payment.voidedAt && payment.status === PaymentStatus.confirmed;

export const paymentFree = (payment: PaymentLike): Money =>
  payment.amount.minus(activeAllocated(payment.allocations));

export const chargeRemaining = (charge: ChargeLike): Money =>
  charge.amount.minus(activeAllocated(charge.allocations));

export const chargeState = (amount: Money, paid: Money): ChargeState => {
  if (paid.isZero()) return 'unpaid';
  return paid.gte(amount) ? 'paid' : 'partial';
};

export function summarize(charges: ChargeLike[], payments: PaymentLike[]): FinanceSummary {
  const counted = payments.filter(isCountedPayment);
  const categories = Object.values(FinanceCategory).map((category) => {
    const own = charges.filter((c) => !c.voidedAt && c.category === category);
    const charged = sumMoney(own.map((c) => c.amount));
    const paid = sumMoney(own.map((c) => activeAllocated(c.allocations)));
    const prepayment = sumMoney(counted.filter((p) => p.category === category).map(paymentFree));
    return { category, charged, paid, debt: charged.minus(paid), prepayment };
  });
  const unallocated = sumMoney(counted.filter((p) => p.category === null).map(paymentFree));
  const totalDue = sumMoney(categories.map((c) => c.debt))
    .minus(sumMoney(categories.map((c) => c.prepayment)))
    .minus(unallocated);

  return {
    totals: { totalDue: formatMoney(totalDue), unallocated: formatMoney(unallocated) },
    categories: categories.map((c) => ({
      category: c.category,
      charged: formatMoney(c.charged),
      paid: formatMoney(c.paid),
      debt: formatMoney(c.debt),
      prepayment: formatMoney(c.prepayment),
    })),
  };
}

const orderKey = (charge: ChargeLike) => (charge.periodStart ?? charge.createdAt).getTime();

export function suggestAllocation(
  charges: ChargeLike[],
  amount: Money,
): { lines: SuggestionLine[]; leftover: string } {
  const open = charges
    .filter((c) => !c.voidedAt)
    .map((charge) => ({ charge, remaining: chargeRemaining(charge) }))
    .filter(({ remaining }) => remaining.gt(0))
    .sort(
      (a, b) =>
        orderKey(a.charge) - orderKey(b.charge) ||
        a.charge.createdAt.getTime() - b.charge.createdAt.getTime(),
    );

  let left = amount;
  const lines = open.map(({ charge, remaining }) => {
    const take = Prisma.Decimal.min(remaining, left);
    left = left.minus(take);
    return {
      chargeId: charge.id,
      category: charge.category,
      description: charge.description,
      periodStart: charge.periodStart,
      createdAt: charge.createdAt,
      remaining: formatMoney(remaining),
      suggestedAmount: formatMoney(take),
    };
  });

  return { lines, leftover: formatMoney(left) };
}
```

```ts
  ChargeLike,
  PaymentLike,

const personSelect = { select: { id: true, firstName: true, lastName: true } } as const;
const allocationAmounts = { select: { amount: true, voidedAt: true } } as const;

export const chargeInclude = {
  allocations: true,

  createdBy: personSelect,
  voidedBy: personSelect,

export type PersonRef = { id: string; firstName: string; lastName: string };

export type FinanceEventType =
  | 'charge_created'
  | 'charge_voided'
  | 'payment_recorded'
  | 'payment_voided'
  | 'allocation_created'
  | 'allocation_voided';

export interface FinanceEvent {
  type: FinanceEventType;
  at: Date;
  by: PersonRef | null;
  amount: string;
  description: string;
  reason: string | null;
}

export interface BalanceRow {

  lastName: string;

  unallocated: string;

export const toChargeView = (charge: LoadedCharge) => {
  const paid = activeAllocated(charge.allocations);
  return {
    id: charge.id,
    category: charge.category,
    amount: formatMoney(charge.amount),
    description: charge.description,
    periodStart: charge.periodStart,
    periodEnd: charge.periodEnd,
    sourceType: charge.sourceType,
    sourceId: charge.sourceId,
    createdAt: charge.createdAt,
    createdBy: charge.createdBy,
    voidedAt: charge.voidedAt,
    voidedBy: charge.voidedBy,
    voidReason: charge.voidReason,
    paid: formatMoney(paid),
    remaining: formatMoney(charge.amount.minus(paid)),
    state: chargeState(charge.amount, paid),
  };
};

export const toPaymentView = (payment: LoadedPayment) => ({
  id: payment.id,
  amount: formatMoney(payment.amount),
  category: payment.category,
  method: payment.method,
  status: payment.status,
  reference: payment.reference,
  receivedAt: payment.receivedAt,

  createdAt: payment.createdAt,
  createdBy: payment.createdBy,
  voidedAt: payment.voidedAt,

  voidReason: payment.voidReason,

export const buildChildFinance = (
  child: PersonRef,
  charges: LoadedCharge[],
  payments: LoadedPayment[],
) => ({
  child,
  ...summarize(charges, payments),
  charges: charges.map(toChargeView),
  payments: payments.map(toPaymentView),
});

export function buildHistory(charges: LoadedCharge[], payments: LoadedPayment[]): FinanceEvent[] {
  const events: FinanceEvent[] = [];

  for (const charge of charges) {
    const amount = formatMoney(charge.amount);
    events.push({
      type: 'charge_created',
      at: charge.createdAt,
      by: charge.createdBy,
      amount,
      description: charge.description,
      reason: null,
    });
    if (charge.voidedAt) {
      events.push({
        type: 'charge_voided',
        at: charge.voidedAt,
        by: charge.voidedBy,
        amount,
        description: charge.description,
        reason: charge.voidReason,
      });
    }
  }

  for (const payment of payments) {
    const amount = formatMoney(payment.amount);
    events.push({
      type: 'payment_recorded',
      at: payment.createdAt,
      by: payment.createdBy,
      amount,
      description: payment.reference,
      reason: null,
    });
    if (payment.voidedAt) {
      events.push({
        type: 'payment_voided',
        at: payment.voidedAt,
        by: payment.voidedBy,
        amount,
        description: payment.reference,
        reason: payment.voidReason,
      });
    }
    for (const allocation of payment.allocations) {
      const description = `${payment.reference} → ${allocation.charge.description}`;
      const allocated = formatMoney(allocation.amount);
      events.push({
        type: 'allocation_created',
        at: allocation.createdAt,
        by: allocation.createdBy,
        amount: allocated,
        description,
        reason: null,
      });
      if (allocation.voidedAt) {
        events.push({
          type: 'allocation_voided',
          at: allocation.voidedAt,
          by: allocation.voidedBy,
          amount: allocated,
          description,
          reason: allocation.voidReason,
        });
      }
    }
  }

  return events.sort((a, b) => b.at.getTime() - a.at.getTime());
}

export function buildBalanceRow(
  child: PersonRef,
  charges: ChargeLike[],
  payments: PaymentLike[],
  category?: FinanceCategory,
): BalanceRow {
  const summary = summarize(charges, payments);
  const categories = category
    ? summary.categories.filter((c) => c.category === category)
    : summary.categories;
  const debt = sumMoney(categories.map((c) => c.debt));
  const prepayment = sumMoney(categories.map((c) => c.prepayment));
  const totalDue = category ? debt.minus(prepayment) : money(summary.totals.totalDue);

  return {
    childId: child.id,
    firstName: child.firstName,
    lastName: child.lastName,
    totalDue: formatMoney(totalDue),
    debt: formatMoney(debt),
    prepayment: formatMoney(prepayment),
    unallocated: summary.totals.unallocated,
  };
}
```

```ts
import { BadRequestException, Injectable, NotFoundException } from '@nestjs/common';
import { ChargeSource } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { CreateChargeDto } from './dto/create-charge.dto';
import { money } from './finance-calc';

@Injectable()
export class ChargesService {
  constructor(private prisma: PrismaService) {}

  async create(childId: string, dto: CreateChargeDto, userId: string) {
    const child = await this.prisma.child.findUnique({ where: { id: childId } });
    if (!child) throw new NotFoundException('Child not found');
    if (money(dto.amount).lte(0)) {
      throw new BadRequestException('Amount must be greater than 0');
    }
    if (dto.periodStart && dto.periodEnd && new Date(dto.periodEnd) < new Date(dto.periodStart)) {
      throw new BadRequestException('Period end must not be before period start');
    }

    return this.prisma.charge.create({
      data: {
        childId,
        category: dto.category,
        amount: dto.amount,
        description: dto.description,
        periodStart: dto.periodStart ? new Date(dto.periodStart) : undefined,
        periodEnd: dto.periodEnd ? new Date(dto.periodEnd) : undefined,
        sourceType: ChargeSource.manual,
        createdById: userId,
      },
    });
  }

  async void(id: string, reason: string, userId: string) {
    return this.prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT id FROM "Charge" WHERE id = ${id} FOR UPDATE`;
      const charge = await tx.charge.findUnique({ where: { id } });
      if (!charge) throw new NotFoundException('Charge not found');
      if (charge.voidedAt) throw new BadRequestException('Charge is voided');

      const data = { voidedAt: new Date(), voidedById: userId, voidReason: reason };
      await tx.allocation.updateMany({ where: { chargeId: id, voidedAt: null }, data });
      return tx.charge.update({ where: { id }, data });
    });
  }
}
```

```ts
import { BadRequestException, NotFoundException } from '@nestjs/common';
import { PaymentStatus, Prisma } from '@prisma/client';
import { activeAllocated, money, sumMoney } from './finance-calc';

export interface AllocationItem {
  chargeId: string;
  amount: string;
}

export async function applyAllocations(
  tx: Prisma.TransactionClient,
  paymentId: string,
  items: AllocationItem[],
  userId: string,
) {
  const chargeIds = items.map((item) => item.chargeId);
  if (new Set(chargeIds).size !== chargeIds.length) {
    throw new BadRequestException('Duplicate charge in allocations');
  }
  if (items.some((item) => money(item.amount).lte(0))) {
    throw new BadRequestException('Amount must be greater than 0');
  }

  await tx.$queryRaw`SELECT id FROM "Payment" WHERE id = ${paymentId} FOR UPDATE`;
  const payment = await tx.payment.findUnique({ where: { id: paymentId } });
  if (!payment) throw new NotFoundException('Payment not found');
  if (payment.voidedAt) throw new BadRequestException('Payment is voided');
  if (payment.status !== PaymentStatus.confirmed) {
    throw new BadRequestException('Payment is not confirmed');
  }

  await tx.$queryRaw`SELECT id FROM "Charge" WHERE id IN (${Prisma.join(chargeIds)}) ORDER BY id FOR UPDATE`;
  const charges = await tx.charge.findMany({
    where: { id: { in: chargeIds } },
    include: { allocations: { where: { voidedAt: null } } },
  });

  for (const item of items) {
    const charge = charges.find((c) => c.id === item.chargeId);
    if (!charge) throw new NotFoundException('Charge not found');
    if (charge.voidedAt) throw new BadRequestException('Charge is voided');
    if (charge.childId !== payment.childId) {
      throw new BadRequestException('Charge belongs to another child');
    }
    if (payment.category && charge.category !== payment.category) {
      throw new BadRequestException('Charge category does not match payment category');
    }
    if (money(item.amount).gt(charge.amount.minus(activeAllocated(charge.allocations)))) {
      throw new BadRequestException('Allocation exceeds charge remaining amount');
    }
  }

  const paymentAllocations = await tx.allocation.findMany({
    where: { paymentId, voidedAt: null },
  });
  const free = payment.amount.minus(activeAllocated(paymentAllocations));
  if (sumMoney(items.map((item) => item.amount)).gt(free)) {
    throw new BadRequestException('Allocation exceeds payment free amount');
  }

  const allocations = [];
  for (const item of items) {
    allocations.push(
      await tx.allocation.create({
        data: { paymentId, chargeId: item.chargeId, amount: item.amount, createdById: userId },
      }),
    );
  }

  return { payment, allocations };
}
```

```ts
import { BadRequestException, Injectable, NotFoundException } from '@nestjs/common';
import { PaymentStatus } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { AllocationItem, applyAllocations } from './allocations';
import { CreatePaymentDto } from './dto/create-payment.dto';
import { money } from './finance-calc';

@Injectable()
export class PaymentsService {
  constructor(private prisma: PrismaService) {}

  async create(childId: string, dto: CreatePaymentDto, userId: string) {
    const child = await this.prisma.child.findUnique({ where: { id: childId } });
    if (!child) throw new NotFoundException('Child not found');
    if (money(dto.amount).lte(0)) {
      throw new BadRequestException('Amount must be greater than 0');
    }

    return this.prisma.$transaction(async (tx) => {
      const payment = await tx.payment.create({
        data: {
          childId,
          amount: dto.amount,
          category: dto.category,
          method: dto.method,
          status: PaymentStatus.confirmed,
          reference: dto.reference,
          receivedAt: new Date(dto.receivedAt),
          note: dto.note,
          createdById: userId,
        },
      });
      if (dto.allocations?.length) {
        await applyAllocations(tx, payment.id, dto.allocations, userId);
      }
      return payment;
    });
  }

  async allocate(paymentId: string, items: AllocationItem[], userId: string) {
    return this.prisma.$transaction(async (tx) => {
      const { payment } = await applyAllocations(tx, paymentId, items, userId);
      return payment;
    });
  }

  async void(id: string, reason: string, userId: string) {
    return this.prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT id FROM "Payment" WHERE id = ${id} FOR UPDATE`;
      const payment = await tx.payment.findUnique({ where: { id } });
      if (!payment) throw new NotFoundException('Payment not found');
      if (payment.voidedAt) throw new BadRequestException('Payment is voided');

      const data = { voidedAt: new Date(), voidedById: userId, voidReason: reason };
      await tx.allocation.updateMany({ where: { paymentId: id, voidedAt: null }, data });
      return tx.payment.update({ where: { id }, data });
    });
  }

  async voidAllocation(id: string, reason: string, userId: string) {
    return this.prisma.$transaction(async (tx) => {
      const allocation = await tx.allocation.findUnique({
        where: { id },
        include: { payment: { select: { childId: true } } },
      });
      if (!allocation) throw new NotFoundException('Allocation not found');

      const { count } = await tx.allocation.updateMany({
        where: { id, voidedAt: null },
        data: { voidedAt: new Date(), voidedById: userId, voidReason: reason },
      });
      if (count === 0) throw new BadRequestException('Allocation is voided');

      return { childId: allocation.payment.childId };
    });
  }
}
```

```ts
import { BadRequestException, Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { AllocationSuggestionDto } from './dto/allocation-suggestion.dto';
import { BalancesQueryDto } from './dto/balances-query.dto';
import { money, suggestAllocation } from './finance-calc';
import {
  PersonRef,
  balanceChildSelect,
  buildBalanceRow,
  buildChildFinance,
  buildHistory,
  chargeInclude,
  paymentInclude,
} from './finance-view';

@Injectable()
export class FinanceService {
  constructor(private prisma: PrismaService) {}

  private async findChild(childId: string): Promise<PersonRef> {
    const child = await this.prisma.child.findUnique({
      where: { id: childId },
      select: { id: true, firstName: true, lastName: true },
    });
    if (!child) throw new NotFoundException('Child not found');
    return child;
  }

  private loadRecords(childId: string) {
    return Promise.all([
      this.prisma.charge.findMany({
        where: { childId },
        include: chargeInclude,
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.payment.findMany({
        where: { childId },
        include: paymentInclude,
        orderBy: { receivedAt: 'desc' },
      }),
    ]);
  }

  async getChildFinance(childId: string) {
    const child = await this.findChild(childId);
    const [charges, payments] = await this.loadRecords(childId);
    return buildChildFinance(child, charges, payments);
  }

  async getHistory(childId: string) {
    await this.findChild(childId);
    const [charges, payments] = await this.loadRecords(childId);
    return buildHistory(charges, payments);
  }

  async suggest(childId: string, dto: AllocationSuggestionDto) {
    await this.findChild(childId);
    const amount = money(dto.amount);
    if (amount.lte(0)) throw new BadRequestException('Amount must be greater than 0');

    const charges = await this.prisma.charge.findMany({
      where: { childId, voidedAt: null, category: dto.category },
      include: { allocations: { where: { voidedAt: null } } },
    });
    return suggestAllocation(charges, amount);
  }

  async getBalances(query: BalancesQueryDto) {
    const children = await this.prisma.child.findMany({
      select: balanceChildSelect,
      orderBy: [{ lastName: 'asc' }, { firstName: 'asc' }],
    });

    return children
      .map((child) => buildBalanceRow(child, child.charges, child.payments, query.category))
      .filter((row) => query.hasDebt !== 'true' || money(row.debt).gt(0))
      .filter((row) => query.hasUnallocated !== 'true' || money(row.unallocated).gt(0));
  }
}
```

The `suggest` test expects `category: FinanceCategory.tuition` in `where`; when `dto.category` is `undefined`, Prisma ignores the key.

```markdown
### Domain: finance

Spec: `docs/superpowers/specs/2026-10-11-finance-design.md`. Per-child ledger in `src/finance/`:

- `Charge` (owed), `Payment` (received), `Allocation` (payment part → charge). Balances are never stored; `finance-calc.ts` computes them, `finance-view.ts` shapes responses.
```

```ts
  history: FinanceEvent[];
  isLoading: boolean;
  isSaving: boolean;
  createChargeModal: { isOpen: boolean };
  createPaymentModal: { isOpen: boolean };
  allocateModal: { isOpen: boolean; payment: FinancePayment | null };
  voidModal: { isOpen: boolean; target: VoidTarget | null };
  setCreateChargeModal: (value: Partial<{ isOpen: boolean }>) => void;
  setCreatePaymentModal: (value: Partial<{ isOpen: boolean }>) => void;
  setAllocateModal: (value: Partial<{ isOpen: boolean; payment: FinancePayment | null }>) => void;
  setVoidModal: (value: Partial<{ isOpen: boolean; target: VoidTarget | null }>) => void;
  fetchFinance: (childId: string) => Promise<void>;
  suggestAllocation: (childId: string, dto: AllocationSuggestionDto) => Promise<AllocationSuggestion>;
  createCharge: (childId: string, dto: CreateChargeDto) => Promise<void>;
  createPayment: (childId: string, dto: CreatePaymentDto) => Promise<void>;
  allocate: (paymentId: string, dto: AllocateDto) => Promise<void>;
  voidRecord: (target: VoidTarget, reason: string) => Promise<void>;

const voidRequests = {
  charge: financeApi.voidCharge,
  payment: financeApi.voidPayment,
  allocation: financeApi.voidAllocation,
};

export const useChildFinanceStore = create<ChildFinanceState>((set) => {
  const applyMutation = async (request: () => Promise<{ data: ChildFinance }>) => {
    set({ isSaving: true });
    try {
      const { data } = await request();
      const { data: history } = await financeApi.getHistory(data.child.id);
      set({ finance: data, history });
    } finally {
      set({ isSaving: false });
    }
  };

  return {
    finance: null,
    history: [],
    isLoading: false,
    isSaving: false,
    createChargeModal: { isOpen: false },
    createPaymentModal: { isOpen: false },
    allocateModal: { isOpen: false, payment: null },
    voidModal: { isOpen: false, target: null },

    setCreateChargeModal: (value) =>
      set((state) => ({ createChargeModal: { ...state.createChargeModal, ...value } })),

    setCreatePaymentModal: (value) =>
      set((state) => ({ createPaymentModal: { ...state.createPaymentModal, ...value } })),

    setAllocateModal: (value) =>
      set((state) => ({ allocateModal: { ...state.allocateModal, ...value } })),

    setVoidModal: (value) => set((state) => ({ voidModal: { ...state.voidModal, ...value } })),

    fetchFinance: async (childId) => {
      set({ isLoading: true });
      try {
        const [{ data: finance }, { data: history }] = await Promise.all([
          financeApi.getChildFinance(childId),
          financeApi.getHistory(childId),
        ]);
        set({ finance, history });
      } finally {
        set({ isLoading: false });
      }
    },

    suggestAllocation: async (childId, dto) => (await financeApi.suggest(childId, dto)).data,

    createCharge: (childId, dto) => applyMutation(() => financeApi.createCharge(childId, dto)),

    createPayment: (childId, dto) => applyMutation(() => financeApi.createPayment(childId, dto)),

    allocate: (paymentId, dto) => applyMutation(() => financeApi.allocate(paymentId, dto)),

    voidRecord: (target, reason) =>
      applyMutation(() => voidRequests[target.kind](target.id, { reason })),
  };
});
```

```tsx
import { useEffect, useState } from 'react'
import { useForm, Controller } from 'react-hook-form'
import { yupResolver } from '@hookform/resolvers/yup'
import * as yup from 'yup'
import { Box, Button, Dialog, DialogTitle, TextField, Typography } from '@mui/material'
import { useChildFinanceStore } from '../model/childFinanceStore'
import { allocationItemSchema } from '../model/schemas'
import { AllocationTable } from './AllocationTable'
import {
  financeCategoryLabels,
  formatMoney,
  toAllocationItems,
  toKopecks,
  type SuggestionLine,
} from '../../../shared/api'

const schema = yup.object({
  allocations: yup.array().of(allocationItemSchema).default([]),
})

type FormValues = yup.InferType<typeof schema>

export const AllocateModal = () => {
  const childId = useChildFinanceStore((s) => s.finance?.child.id)
  const { isOpen, payment } = useChildFinanceStore((s) => s.allocateModal)
```
