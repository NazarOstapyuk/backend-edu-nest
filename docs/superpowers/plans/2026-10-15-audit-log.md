```ts
import { Prisma } from '@prisma/client';

export type FieldChange = {
  field: string;
  before: string | null;
  after: string | null;
};

const SECRET_KEYS = new Set([
  'password',
  'currentPassword',
  'newPassword',
  'refreshToken',
  'accessToken',
  'token',
]);
const MAX_BODY = 10_000;
const MUTATING = new Set(['POST', 'PUT', 'PATCH', 'DELETE']);
const SKIPPED = new Set([
  '/auth/login',
  '/auth/refresh',
  '/auth/logout',
  '/me/password',
]);

export function normalizeValue(value: unknown): string | null {
  if (value === undefined || value === null) return null;
  if (value instanceof Prisma.Decimal) return value.toFixed(2);
  if (value instanceof Date) {
    const iso = value.toISOString();
    return iso.endsWith('T00:00:00.000Z') ? iso.slice(0, 10) : iso;
  }
  if (Array.isArray(value))
    return value
      .map((item) => normalizeValue(item) ?? '')
      .sort()
      .join(', ');
  if (typeof value === 'object') return JSON.stringify(value);
  return String(value);
}

export function diffFields(
  before: Record<string, unknown> | null,
  after: Record<string, unknown> | null,
  fields: readonly string[],
): FieldChange[] {
  return fields.flatMap((field) => {
    const was = before ? normalizeValue(before[field]) : null;
    const now = after ? normalizeValue(after[field]) : null;
    return was === now ? [] : [{ field, before: was, after: now }];
  });
}

const strip = (value: unknown): unknown => {
  if (Array.isArray(value)) return value.map(strip);
  if (value && typeof value === 'object' && !(value instanceof Date))
    return Object.fromEntries(
      Object.entries(value)
        .filter(([key]) => !SECRET_KEYS.has(key))
        .map(([key, item]) => [key, strip(item)]),
    );
  return value;
};

export function sanitizeBody(body: unknown): Prisma.InputJsonValue | null {
  if (body === undefined || body === null) return null;
  const clean = strip(body);
  const json = JSON.stringify(clean);
  if (json === undefined) return null;
  if (json.length > MAX_BODY) return { truncated: true, size: json.length };
  return JSON.parse(json) as Prisma.InputJsonValue;
}

export function entityFromRoute(
  route: string,
  params: Record<string, string>,
  responseId: string | null,
): { entityType: string | null; entityId: string | null } {
  const path = route.slice(route.indexOf(' ') + 1);
  const entityType = path.split('/').filter(Boolean)[0] ?? null;
  const entityId =
    params.id ?? params.childId ?? params.teacherId ?? responseId ?? null;
  return { entityType, entityId };
}

export function shouldRecordEvent(method: string, path: string): boolean {
  if (!MUTATING.has(method.toUpperCase())) return false;
  return !SKIPPED.has(path.split('?')[0]);
}

export function kyivDayStart(day: string): Date {
  const utcMidnight = new Date(`${day}T00:00:00.000Z`);
  const zone =
    new Intl.DateTimeFormat('en-US', {
      timeZone: 'Europe/Kyiv',
      timeZoneName: 'shortOffset',
    })
      .formatToParts(utcMidnight)
      .find((part) => part.type === 'timeZoneName')?.value ?? 'GMT+2';
  const hours = Number(zone.replace('GMT', '') || '0');
  return new Date(utcMidnight.getTime() - hours * 3_600_000);
}
```

```ts
interface RequestStore {

  responseId: string | null;
}

const storage = new AsyncLocalStorage<RequestStore>();

export const runWithRequest = <T>(requestId: string, fn: () => T): T =>

export const currentRequestId = (): string | null =>
  storage.getStore()?.requestId ?? null;

export const currentResponseId = (): string | null =>
  storage.getStore()?.responseId ?? null;

export const setResponseId = (id: string): void => {
  const store = storage.getStore();
  if (store) store.responseId = id;
};
```

```ts
import {
  BadRequestException,
  Injectable,
  Logger,
  NotFoundException,
} from '@nestjs/common';
import { AuditAction, AuthEventType, Prisma, Role } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { parseDay } from '../classes/access';
import { addDays, isoDay } from '../tariffs/dates';
import { diffFields, kyivDayStart } from './audit-calc';
import { currentRequestId } from './request-context';
import {
  AuditChangesQueryDto,
  AuditEventsQueryDto,
  AuthEventsQueryDto,
} from './dto/audit-query.dto';

const PAGE = 50;
const person = { select: { id: true, firstName: true, lastName: true, login: true } };

type Db = Prisma.TransactionClient | PrismaService;

@Injectable()
export class AuditService {
  private readonly logger = new Logger(AuditService.name);

  constructor(private prisma: PrismaService) {}

  async recordEvent(input: {
    requestId: string;
    actorId: string | null;
    actorRole: Role | null;
    method: string;
    route: string;
    entityType: string | null;
    entityId: string | null;
    statusCode: number;
    ip: string | null;
    userAgent: string | null;
    body: Prisma.InputJsonValue | null;
  }): Promise<void> {
    try {
      await this.prisma.auditEvent.create({
        data: { ...input, body: input.body ?? Prisma.JsonNull },
      });
    } catch (error) {
      this.logger.error(`Audit event not recorded: ${String(error)}`);
    }
  }

  async recordAuth(input: {
    type: AuthEventType;
    login: string;
    userId: string | null;
    reason: string | null;
    ip: string | null;
    userAgent: string | null;
  }): Promise<void> {
    try {
      await this.prisma.authEvent.create({
        data: { ...input, login: input.login.slice(0, 100) },
      });
    } catch (error) {
      this.logger.error(`Auth event not recorded: ${String(error)}`);
    }
  }

  async recordChange(
    db: Db,
    input: {
      actorId: string | null;
      entityType: string;
      entityId: string;
      action: AuditAction;
      before: Record<string, unknown> | null;
      after: Record<string, unknown> | null;
      fields: readonly string[];
    },
  ): Promise<void> {
    const changes = diffFields(input.before, input.after, input.fields);
    if (input.action === AuditAction.update && changes.length === 0) return;
    await db.auditChange.create({
      data: {
        requestId: currentRequestId() ?? '',
        actorId: input.actorId,
        entityType: input.entityType,
        entityId: input.entityId,
        action: input.action,
        changes,
      },
    });
  }

  private period(from?: string, to?: string) {
    if (!from && !to) return undefined;
    const start = from ? parseDay(from) : null;
    const end = to ? parseDay(to) : null;
    if (start && end && end.getTime() < start.getTime())
      throw new BadRequestException('Invalid period');
    return {
      ...(start ? { gte: kyivDayStart(isoDay(start)) } : {}),
      ...(end ? { lt: kyivDayStart(isoDay(addDays(end, 1))) } : {}),
    };
  }

  private paging(page?: number) {
    const current = page ?? 1;
    return { current, skip: (current - 1) * PAGE, take: PAGE };
  }

  async listEvents(query: AuditEventsQueryDto) {
    const at = this.period(query.from, query.to);
    const where: Prisma.AuditEventWhereInput = {
      ...(at ? { at } : {}),
      ...(query.actorId ? { actorId: query.actorId } : {}),
      ...(query.entityType ? { entityType: query.entityType } : {}),
      ...(query.entityId ? { entityId: query.entityId } : {}),
      ...(query.failedOnly === 'true' ? { statusCode: { gte: 400 } } : {}),
    };
    const { current, skip, take } = this.paging(query.page);
    const [items, total] = await this.prisma.$transaction([
      this.prisma.auditEvent.findMany({
        where,
        include: { actor: person },
        orderBy: { at: 'desc' },
        skip,
        take,
      }),
      this.prisma.auditEvent.count({ where }),
    ]);
    return { items, total, page: current };
  }

  async getEvent(id: string) {
    const event = await this.prisma.auditEvent.findUnique({
      where: { id },
      include: { actor: person },
    });
    if (!event) throw new NotFoundException('Audit event not found');
    const changes = await this.prisma.auditChange.findMany({
      where: { requestId: event.requestId },
      orderBy: { at: 'asc' },
    });
    return { ...event, changes };
  }

  async listChanges(query: AuditChangesQueryDto) {
    const where = { entityType: query.entityType, entityId: query.entityId };
    const { current, skip, take } = this.paging(query.page);
    const [items, total] = await this.prisma.$transaction([
      this.prisma.auditChange.findMany({
        where,
        include: { actor: person },
        orderBy: { at: 'desc' },
        skip,
        take,
      }),
      this.prisma.auditChange.count({ where }),
    ]);
    return { items, total, page: current };
  }

  async listAuth(query: AuthEventsQueryDto) {
    const at = this.period(query.from, query.to);
    const where: Prisma.AuthEventWhereInput = {
      ...(at ? { at } : {}),
      ...(query.userId ? { userId: query.userId } : {}),
      ...(query.login
        ? { login: { contains: query.login, mode: Prisma.QueryMode.insensitive } }
        : {}),
      ...(query.type ? { type: query.type } : {}),
    };
    const { current, skip, take } = this.paging(query.page);
    const [items, total] = await this.prisma.$transaction([
      this.prisma.authEvent.findMany({
        where,
        include: { user: person },
        orderBy: { at: 'desc' },
        skip,
        take,
      }),
      this.prisma.authEvent.count({ where }),
    ]);
    return { items, total, page: current };
  }
}
```

```ts
import { Injectable, NestMiddleware } from '@nestjs/common';
import type { NextFunction, Request, Response } from 'express';
import { randomUUID } from 'node:crypto';
import type { AuthenticatedUser } from '../common/types/authenticated-user.type';
import { entityFromRoute, sanitizeBody, shouldRecordEvent } from './audit-calc';
import { AuditService } from './audit.service';
import { currentResponseId, runWithRequest } from './request-context';
import { requestMeta } from './request-meta';

type RoutedRequest = Request & {
  route?: { path?: string };
  user?: AuthenticatedUser;
};

@Injectable()
export class AuditMiddleware implements NestMiddleware {
  constructor(private audit: AuditService) {}

  use(req: RoutedRequest, res: Response, next: NextFunction) {
    const requestId = randomUUID();
    runWithRequest(requestId, () => {
      res.on('finish', () => {
        if (!shouldRecordEvent(req.method, req.originalUrl)) return;
        const path = req.route?.path
          ? `${req.baseUrl}${req.route.path}`
          : req.originalUrl.split('?')[0];
        const route = `${req.method} ${path}`;
        const params = (req.params ?? {}) as Record<string, string>;
        const { entityType, entityId } = entityFromRoute(
          route,
          params,
          currentResponseId(),
        );
        const meta = requestMeta(req);
        void this.audit.recordEvent({
          requestId,
          actorId: req.user?.id ?? null,
          actorRole: req.user?.role ?? null,
          method: req.method,
          route,
          entityType,
          entityId,
          statusCode: res.statusCode,
          ip: meta.ip,
          userAgent: meta.userAgent,
          body: sanitizeBody(req.body),
        });
      });
      next();
    });
  }
}
```

`currentResponseId()` inside the `finish` callback: the callback is registered inside `runWithRequest`, so it runs in the same async context. Verify with the test below; if `res.on('finish')` loses the store on this platform, capture a reference to the store object instead (export `currentStore()` from `request-context.ts`) and read `store.responseId` in the callback. `req.params` may be empty at middleware level in Express 5 (params are set per matched route); if a test against a running app (Task 7 live check) shows `entityId` missing for `PATCH /children/:id`, move the params capture into `AuditResponseInterceptor` (store `params` in the request store alongside `responseId`).

```ts
import { Observable, tap } from 'rxjs';

@Injectable()
export class AuditResponseInterceptor implements NestInterceptor {
  intercept(_context: ExecutionContext, next: CallHandler): Observable<unknown> {
    return next.handle().pipe(
      tap((data: unknown) => {
        if (data && typeof data === 'object' && 'id' in data) {
          const id = (data as { id: unknown }).id;
          if (typeof id === 'string') setResponseId(id);
        }
      }),
    );
  }
}
```

```ts
import { PrismaService } from '../prisma/prisma.service';

const KEEP_MS = 365 * 24 * 60 * 60 * 1000;

@Injectable()
export class AuditRetentionScheduler {
  private readonly logger = new Logger(AuditRetentionScheduler.name);

  constructor(private prisma: PrismaService) {}

  @Cron('0 4 * * *', { timeZone: 'Europe/Kyiv' })
  async purge(now: Date = new Date()) {
    const before = new Date(now.getTime() - KEEP_MS);
    try {
      const [events, changes, auth] = await Promise.all([

      ]);
      this.logger.log(`Audit purge: ${events.count} events, ${changes.count} changes, ${auth.count} logins`);
    } catch (error) {
      this.logger.error(`Audit purge failed: ${String(error)}`);
    }
  }
}
```
