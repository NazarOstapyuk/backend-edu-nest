# Meals Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Admins build menus per date for kindergarten and school (from cyclic templates), price complexes per price category, and publish; parents order a complex per day (with dish choices) before a daily deadline; every order is a `meals` charge in the finance ledger; teachers see today's orders of their groups.

**Architecture:** New backend module `src/meals/` with a pure `meals-calc.ts` (Kyiv deadline, price on date, dish choice resolution, template week, copy offset), directory services, a `MenusService` (menu days + templates), an `OrdersService` (create/cancel with child-row lock and charge creation, plus `cancelOrdersForNoticeInTx` used by absence-notice approval) and a `MealViewsService` (admin list + summary, teacher today, parent menu). The frontend adds a «Харчування» admin page (menu, templates, directories, orders), a teacher block and a parent tab.

**Tech Stack:** NestJS 11, Prisma 7, class-validator/class-transformer, Jest. Frontend: React 19, MUI 9, Zustand 5, react-hook-form 7 + yup, axios.

```ts
import { addDays, dayCount } from '../tariffs/dates';
import { isoWeekday } from '../timetable/schedule-calc';

export interface PriceLike {
  complexId: string;
  priceCategoryId: string;
  validFrom: Date;
}

export interface DishLike {
  id: string;
  courseId: string;
}

export type ChoiceError = 'course_missing' | 'choice_required' | 'invalid_choice';

export type ChoiceResult =
  | { items: { courseId: string; dishId: string }[] }
  | { error: ChoiceError };

export const choiceErrorMessage: Record<ChoiceError, string> = {
  course_missing: 'Complex is not available on this day',
  choice_required: 'Dish choice is required',
  invalid_choice: 'Invalid dish choice',
};

const MINUTE_MS = 60 * 1000;

export function parseTime(value: string): number | null {
  const match = /^([01]\d|2[0-3]):([0-5]\d)$/.exec(value);
  if (!match) return null;
  return Number(match[1]) * 60 + Number(match[2]);
}

const kyivOffsetMinutes = (instant: Date): number => {
  const parts = new Intl.DateTimeFormat('en-US', {
    timeZone: 'Europe/Kyiv',
    hourCycle: 'h23',
    year: 'numeric',
    month: '2-digit',
    day: '2-digit',
    hour: '2-digit',
    minute: '2-digit',
  }).formatToParts(instant);
  const get = (type: string) =>
    Number(parts.find((p) => p.type === type)?.value);
  const asUtc = Date.UTC(
    get('year'),
    get('month') - 1,
    get('day'),
    get('hour'),
    get('minute'),
  );
  return Math.round((asUtc - instant.getTime()) / MINUTE_MS);
};

export function deadlineFor(date: Date, hhmm: string): Date {
  const minutes = parseTime(hhmm) ?? 0;
  const guess = new Date(date.getTime() + minutes * MINUTE_MS);
  return new Date(guess.getTime() - kyivOffsetMinutes(guess) * MINUTE_MS);
}

export const isBeforeDeadline = (
  date: Date,
  hhmm: string,
  now: Date,
): boolean => now.getTime() < deadlineFor(date, hhmm).getTime();

export function priceOn<T extends PriceLike>(
  prices: T[],
  complexId: string,
  categoryId: string,
  date: Date,
): T | null {
  let best: T | null = null;
  for (const price of prices) {
    if (
      price.complexId !== complexId ||
      price.priceCategoryId !== categoryId ||
      price.validFrom.getTime() > date.getTime()
    )
      continue;
    if (!best || price.validFrom.getTime() > best.validFrom.getTime())
      best = price;
  }
  return best;
}

export function resolveChoices(
  courseIds: string[],
  dishes: DishLike[],
  picked: string[],
): ChoiceResult {
  const byId = new Map(dishes.map((d) => [d.id, d]));
  const pickedByCourse = new Map<string, string>();
  for (const id of picked) {
    const dish = byId.get(id);
    if (
      !dish ||
      !courseIds.includes(dish.courseId) ||
      pickedByCourse.has(dish.courseId)
    )
      return { error: 'invalid_choice' };
    pickedByCourse.set(dish.courseId, id);
  }
  const items: { courseId: string; dishId: string }[] = [];
  for (const courseId of courseIds) {
    const options = dishes.filter((d) => d.courseId === courseId);
    if (!options.length) return { error: 'course_missing' };
    const pickedId = pickedByCourse.get(courseId);
    if (pickedId) items.push({ courseId, dishId: pickedId });
    else if (options.length === 1)
      items.push({ courseId, dishId: options[0].id });
    else return { error: 'choice_required' };
  }
  return { items };
}

export const mondayOf = (day: Date): Date => addDays(day, 1 - isoWeekday(day));

export function templateWeek(
  from: Date,
  startWeek: number,
  weeks: number,
  day: Date,
): number {
  const diff = Math.round(
    (mondayOf(day).getTime() - mondayOf(from).getTime()) /
      (7 * 24 * 60 * MINUTE_MS),
  );
  return ((((startWeek - 1 + diff) % weeks) + weeks) % weeks) + 1;
}

export const copyTarget = (fromStart: Date, toStart: Date, day: Date): Date =>
  addDays(toStart, dayCount(fromStart, day) - 1);
```

```ts
  ConflictException,
  Injectable,
  NotFoundException,

import { PrismaService } from '../prisma/prisma.service';
import {
  CreateDishDto,
  CreateSortedNamedDto,
  DishesQueryDto,
  UpdateDishDto,
  UpdateSortedNamedDto,
} from './dto/meal-directory.dto';

const sorted = [{ sortOrder: 'asc' as const }, { name: 'asc' as const }];

@Injectable()
export class MealDirectoriesService {
  constructor(private prisma: PrismaService) {}

  listCourses() {
    return this.prisma.mealCourse.findMany({ orderBy: sorted });
  }

  async createCourse(dto: CreateSortedNamedDto) {
    const name = dto.name.trim();
    if (await this.prisma.mealCourse.findUnique({ where: { name } }))
      throw new ConflictException('Meal course already exists');
    return this.prisma.mealCourse.create({
      data: { name, sortOrder: dto.sortOrder },
    });
  }

  async updateCourse(id: string, dto: UpdateSortedNamedDto) {
    const course = await this.prisma.mealCourse.findUnique({ where: { id } });
    if (!course) throw new NotFoundException('Meal course not found');
    const name = dto.name?.trim();
    if (
      name &&
      name !== course.name &&
      (await this.prisma.mealCourse.findUnique({ where: { name } }))
    )
      throw new ConflictException('Meal course already exists');
    return this.prisma.mealCourse.update({
      where: { id },
      data: { name, sortOrder: dto.sortOrder, active: dto.active },
    });
  }

  listDishes(query: DishesQueryDto) {
    return this.prisma.dish.findMany({
      where: { courseId: query.courseId, active: query.active },
      include: { course: { select: { id: true, name: true, sortOrder: true } } },
      orderBy: [{ course: { sortOrder: 'asc' } }, { name: 'asc' }],
    });
  }

  async createDish(dto: CreateDishDto) {
    const course = await this.prisma.mealCourse.findUnique({
      where: { id: dto.courseId },
    });
    if (!course) throw new NotFoundException('Meal course not found');
    if (!course.active) throw new BadRequestException('Meal course is inactive');
    const name = dto.name.trim();
    if (
      await this.prisma.dish.findUnique({
        where: { courseId_name: { courseId: dto.courseId, name } },
      })
```

```ts
@Injectable()

  async getDeadline(): Promise<string> {
    const setting = await this.prisma.appSetting.findUnique({
      where: { key: DEADLINE_KEY },
    });
    return setting && parseTime(setting.value) !== null
      ? setting.value
      : DEFAULT_DEADLINE;
  }

  async setDeadline(value: string) {
    if (parseTime(value) === null)
      throw new BadRequestException('Invalid deadline');
    await this.prisma.appSetting.upsert({
      where: { key: DEADLINE_KEY },
      create: { key: DEADLINE_KEY, value },
      update: { value },
    });
    return { mealOrderDeadline: value };
  }
}
```

```ts
import {
  BadRequestException,
  ConflictException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { parseDay } from '../classes/access';
import { formatMoney, money } from '../finance/finance-calc';
import { isoDay, kyivToday } from '../tariffs/dates';
import {
  AddComplexPriceDto,
  CreateMealComplexDto,
  UpdateMealComplexDto,
} from './dto/meal-complex.dto';

const complexInclude = {
  courses: {
    select: { course: { select: { id: true, name: true, sortOrder: true } } },
  },
  prices: {
    select: {
      id: true,
      priceCategoryId: true,
      priceCategory: { select: { name: true } },
      price: true,
      validFrom: true,
    },
    orderBy: [{ validFrom: 'desc' as const }],
  },
} as const;

type ComplexRow = Prisma.MealComplexGetPayload<{
  include: typeof complexInclude;
}>;

export const toComplexView = (row: ComplexRow) => ({
  id: row.id,
  name: row.name,
  description: row.description,
  sortOrder: row.sortOrder,
  active: row.active,
  courses: row.courses
    .map((c) => c.course)
    .sort((a, b) => a.sortOrder - b.sortOrder || a.name.localeCompare(b.name)),
  prices: row.prices.map((p) => ({
    id: p.id,
    priceCategoryId: p.priceCategoryId,
    priceCategoryName: p.priceCategory.name,
    price: formatMoney(p.price),
    validFrom: isoDay(p.validFrom),
  })),
});

@Injectable()
export class MealComplexesService {
  constructor(private prisma: PrismaService) {}

  async list() {
    const rows = await this.prisma.mealComplex.findMany({
      include: complexInclude,
      orderBy: [{ sortOrder: 'asc' }, { name: 'asc' }],
    });
    return rows.map(toComplexView);
  }

  private async view(id: string) {
    const row = await this.prisma.mealComplex.findUnique({
      where: { id },
      include: complexInclude,
    });
    if (!row) throw new NotFoundException('Meal complex not found');
    return toComplexView(row);
  }

  private async assertCourses(ids: string[], currentIds: string[]) {
    const courses = await this.prisma.mealCourse.findMany({
      where: { id: { in: ids } },
      select: { id: true, active: true },
    });
    if (courses.length !== ids.length)
      throw new NotFoundException('Meal course not found');
    if (courses.some((c) => !c.active && !currentIds.includes(c.id)))
      throw new BadRequestException('Meal course is inactive');
  }

  async create(dto: CreateMealComplexDto) {
    const name = dto.name.trim();
    if (await this.prisma.mealComplex.findUnique({ where: { name } }))
      throw new ConflictException('Meal complex already exists');
    await this.assertCourses(dto.courseIds, []);
    const created = await this.prisma.mealComplex.create({
      data: {
        name,
        description: dto.description?.trim() || null,
        sortOrder: dto.sortOrder,
        courses: { create: dto.courseIds.map((courseId) => ({ courseId })) },
      },
    });
    return this.view(created.id);
  }

  async update(id: string, dto: UpdateMealComplexDto) {
    const current = await this.view(id);
    const name = dto.name?.trim();
    if (
      name &&
      name !== current.name &&
      (await this.prisma.mealComplex.findUnique({ where: { name } }))
    )
      throw new ConflictException('Meal complex already exists');
    if (dto.courseIds)
      await this.assertCourses(
        dto.courseIds,
        current.courses.map((c) => c.id),
      );
    const description =
      dto.description === undefined
        ? undefined
        : dto.description?.trim() || null;
    const courseIds = dto.courseIds;
    await this.prisma.$transaction(async (tx) => {
      await tx.mealComplex.update({
        where: { id },
        data: {
          name,
          description,
          sortOrder: dto.sortOrder,
          active: dto.active,
        },
      });
      if (courseIds) {
        await tx.mealComplexCourse.deleteMany({ where: { complexId: id } });
        await tx.mealComplexCourse.createMany({
          data: courseIds.map((courseId) => ({ complexId: id, courseId })),
        });
      }
    });
    return this.view(id);
  }

  async addPrice(id: string, dto: AddComplexPriceDto) {
    if (money(dto.price).lte(0))
      throw new BadRequestException('Amount must be greater than 0');
    await this.view(id);
    const category = await this.prisma.mealPriceCategory.findUnique({
      where: { id: dto.priceCategoryId },
    });
    if (!category) throw new NotFoundException('Meal price category not found');
    if (!category.active)
      throw new BadRequestException('Meal price category is inactive');
    const validFrom = parseDay(dto.validFrom.slice(0, 10));
    const existing = await this.prisma.mealComplexPrice.findUnique({
      where: {
        complexId_priceCategoryId_validFrom: {
          complexId: id,
          priceCategoryId: dto.priceCategoryId,
          validFrom,
        },
      },
    });
    if (existing)
      throw new BadRequestException('Price for this date already exists');
    await this.prisma.mealComplexPrice.create({
      data: {
        complexId: id,
        priceCategoryId: dto.priceCategoryId,
        price: dto.price,
        validFrom,
      },
    });
    return this.view(id);
  }

  async deletePrice(id: string, priceId: string, today: Date = kyivToday()) {
    const price = await this.prisma.mealComplexPrice.findUnique({
      where: { id: priceId },
    });
    if (!price || price.complexId !== id)
      throw new NotFoundException('Price not found');
    if (price.validFrom.getTime() <= today.getTime())
      throw new BadRequestException('Price is already in effect');
    await this.prisma.mealComplexPrice.delete({ where: { id: priceId } });
    return this.view(id);
  }
}
```

- Produces:
  - `MenusService.workingFor(kind: GroupKind, entries: CalendarEntryLike[]): (day: Date) => { working: boolean; title: string | null }` — calendar levels `all` + kind only.
  - `MenusService.fill(kind: GroupKind, plan: { date: Date; dishIds: string[] }[], overwrite: boolean): Promise<FillResult>`, `FillResult = { created: number; overwritten: number; skipped: number }`.
  - `parsePeriod(fromValue: string, toValue: string, maxDays: number): { from: Date; to: Date }` exported from `menus.service.ts` (400 `Invalid period` / `Period is too long`); reused by Task 6.
  - `MenuDayView`:
    ```ts
    interface MenuDayView {
      date: string; working: boolean; title: string | null;
      menu: null | {
        id: string; status: 'draft' | 'published'; publishedAt: Date | null; activeOrders: number;
        dishes: { id: string; name: string; active: boolean; course: { id: string; name: string; sortOrder: number } }[];
      };
    }
    ```
- Routes (`@Roles(Role.root, Role.admin)`):
  - `GET /menu-days?kind&from&to` → `MenuDayView[]` for every date in the period (≤ 62 days)
  - `PUT /menu-days/:date/:kind` `{ dishIds }` → `MenuDayView`
  - `POST /menu-days/publish` / `POST /menu-days/unpublish` `{ kind, from, to }` → `{ count }` (≤ 92 days)
  - `POST /menu-days/copy` `{ kind, fromStart, fromEnd, toStart, overwrite? }` → `FillResult` (source ≤ 92 days)
  - `DELETE /menu-days/:date/:kind` → `{ message: 'Menu day deleted' }`
  - `GET/POST /menu-templates`, `PATCH /menu-templates/:id`, `PUT /menu-templates/:id/items`, `POST /menu-templates/:id/apply` `{ kind, from, to, startWeek, overwrite? }` → `FillResult`

Locking: `PUT` and `unpublish` lock the `MenuDay` rows with `SELECT ... FOR UPDATE` inside the transaction before checking orders; order creation (Task 5) takes `FOR SHARE` on the same row, so a dish cannot be removed while an order for it is being created.

Copy and apply ignore inactive dishes from the source/template; a day left with no dishes is not written and not counted. `overwrite` replaces only draft days (a draft never has active orders because unpublishing requires none).

```ts
import {
  BadRequestException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { CalendarScope, GroupKind, MenuStatus, Prisma } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import { CalendarService } from '../attendance/calendar.service';
import {
  type CalendarEntryLike,
  dayInfo,
  dayRange,
} from '../attendance/register-calc';
import { parseDay } from '../classes/access';
import { dayCount, isoDay } from '../tariffs/dates';
import { copyTarget } from './meals-calc';
import { CopyMenuDto, MenuPeriodDto, SaveMenuDayDto } from './dto/menu.dto';

export interface FillResult {
  created: number;
  overwritten: number;
  skipped: number;
}

export function parsePeriod(
  fromValue: string,
  toValue: string,
  maxDays: number,
): { from: Date; to: Date } {
  const from = parseDay(fromValue);
  const to = parseDay(toValue);
  if (to.getTime() < from.getTime())
    throw new BadRequestException('Invalid period');
  if (dayCount(from, to) > maxDays)
    throw new BadRequestException('Period is too long');
  return { from, to };
}

const dishSelect = {
  id: true,
  name: true,
  active: true,
  course: { select: { id: true, name: true, sortOrder: true } },
} as const;

const sortDishes = <
  T extends { name: string; course: { sortOrder: number; name: string } },
>(
  dishes: T[],
) =>
  [...dishes].sort(
    (a, b) =>
      a.course.sortOrder - b.course.sortOrder ||
      a.course.name.localeCompare(b.course.name) ||
      a.name.localeCompare(b.name),
  );

@Injectable()
export class MenusService {
  constructor(
    private prisma: PrismaService,
    private calendar: CalendarService,
  ) {}

  workingFor(kind: GroupKind, entries: CalendarEntryLike[]) {
    const scoped = entries.filter((e) => e.scope !== CalendarScope.group);
    return (day: Date) => dayInfo(day, { id: '', kind }, scoped);
  }

  async list(query: MenuPeriodDto) {
    const { from, to } = parsePeriod(query.from, query.to, 62);
    return this.views(query.kind, from, to);
  }

  private async views(kind: GroupKind, from: Date, to: Date) {
    const [entries, days] = await Promise.all([
      this.calendar.entriesBetween(from, to),
      this.prisma.menuDay.findMany({
        where: { kind, date: { gte: from, lte: to } },
        include: { items: { select: { dish: { select: dishSelect } } } },
      }),
    ]);
    const counts = days.length
      ? await this.prisma.mealOrder.groupBy({
          by: ['menuDayId'],
          where: {
            menuDayId: { in: days.map((d) => d.id) },
            cancelledAt: null,
          },
          _count: { _all: true },
        })
      : [];
    const working = this.workingFor(kind, entries);
    return dayRange(from, to).map((date) => {
      const info = working(date);
      const menu = days.find((d) => d.date.getTime() === date.getTime());
      return {
        date: isoDay(date),
        working: info.working,
        title: info.title,
        menu: menu
          ? {
              id: menu.id,
              status: menu.status,
              publishedAt: menu.publishedAt,
              activeOrders:
                counts.find((c) => c.menuDayId === menu.id)?._count._all ?? 0,
              dishes: sortDishes(menu.items.map((i) => i.dish)),
            }
          : null,
      };
    });
  }

  async saveDay(dateValue: string, kind: GroupKind, dto: SaveMenuDayDto) {
    const date = parseDay(dateValue);
    const entries = await this.calendar.entriesBetween(date, date);
    if (!this.workingFor(kind, entries)(date).working)
      throw new BadRequestException('Day is not a working day');
    const dishes = dto.dishIds.length
      ? await this.prisma.dish.findMany({
          where: { id: { in: dto.dishIds } },
          select: { id: true, active: true },
        })
      : [];
    if (dishes.length !== dto.dishIds.length)
      throw new NotFoundException('Dish not found');
    await this.prisma.$transaction(async (tx) => {
      const existing = await tx.menuDay.findUnique({
        where: { date_kind: { date, kind } },
        include: { items: { select: { dishId: true } } },
      });
      if (existing)
        await tx.$queryRaw`SELECT id FROM "MenuDay" WHERE id = ${existing.id} FOR UPDATE`;
      const current = existing?.items.map((i) => i.dishId) ?? [];
      if (dishes.some((d) => !d.active && !current.includes(d.id)))
        throw new BadRequestException('Dish is inactive');
      const removed = current.filter((id) => !dto.dishIds.includes(id));
      if (existing && removed.length) {
        const used = await tx.mealOrderItem.count({
          where: {
            dishId: { in: removed },
            order: { menuDayId: existing.id, cancelledAt: null },
          },
        });
        if (used) throw new BadRequestException('Dish is ordered');
      }
      const day = await tx.menuDay.upsert({
        where: { date_kind: { date, kind } },
        create: { date, kind },
        update: {},
      });
      await tx.menuDayItem.deleteMany({ where: { menuDayId: day.id } });
      await tx.menuDayItem.createMany({
        data: dto.dishIds.map((dishId) => ({ menuDayId: day.id, dishId })),
      });
    });
    const [view] = await this.views(kind, date, date);
    return view;
  }

  async publish(dto: MenuPeriodDto) {
    const { from, to } = parsePeriod(dto.from, dto.to, 92);
    const { count } = await this.prisma.menuDay.updateMany({
      where: {
        kind: dto.kind,
        date: { gte: from, lte: to },
        status: MenuStatus.draft,
        items: { some: {} },
      },
      data: { status: MenuStatus.published, publishedAt: new Date() },
    });
    return { count };
  }

  async unpublish(dto: MenuPeriodDto) {
    const { from, to } = parsePeriod(dto.from, dto.to, 92);
    return this.prisma.$transaction(async (tx) => {
      const days = await tx.menuDay.findMany({
        where: {
          kind: dto.kind,
          date: { gte: from, lte: to },
          status: MenuStatus.published,
        },
        select: { id: true },
      });
      if (!days.length) return { count: 0 };
      const ids = days.map((d) => d.id);
      await tx.$queryRaw`SELECT id FROM "MenuDay" WHERE id IN (${Prisma.join(ids)}) ORDER BY id FOR UPDATE`;
      const orders = await tx.mealOrder.count({
        where: { menuDayId: { in: ids }, cancelledAt: null },
      });
      if (orders) throw new BadRequestException('Menu has orders');
      const { count } = await tx.menuDay.updateMany({
        where: { id: { in: ids } },
        data: { status: MenuStatus.draft, publishedAt: null },
      });
      return { count };
    });
  }

  async fill(
    kind: GroupKind,
    plan: { date: Date; dishIds: string[] }[],
    overwrite: boolean,
  ): Promise<FillResult> {
    const result: FillResult = { created: 0, overwritten: 0, skipped: 0 };
    if (!plan.length) return result;
    const times = plan.map((p) => p.date.getTime());
    const from = new Date(Math.min(...times));
    const to = new Date(Math.max(...times));
    const [entries, existing] = await Promise.all([
      this.calendar.entriesBetween(from, to),
      this.prisma.menuDay.findMany({
        where: { kind, date: { gte: from, lte: to } },
        select: { id: true, date: true, status: true },
      }),
    ]);
    const working = this.workingFor(kind, entries);
    await this.prisma.$transaction(async (tx) => {
      for (const item of plan) {
        if (!working(item.date).working) {
          result.skipped++;
          continue;
        }
        const current = existing.find(
          (e) => e.date.getTime() === item.date.getTime(),
        );
        if (current) {
          if (!overwrite || current.status === MenuStatus.published) {
            result.skipped++;
            continue;
          }
          await tx.menuDayItem.deleteMany({ where: { menuDayId: current.id } });
          await tx.menuDayItem.createMany({
            data: item.dishIds.map((dishId) => ({
              menuDayId: current.id,
              dishId,
            })),
          });
          result.overwritten++;
        } else {
          await tx.menuDay.create({
            data: {
              date: item.date,
              kind,
              items: { create: item.dishIds.map((dishId) => ({ dishId })) },
            },
          });
          result.created++;
        }
      }
    });
    return result;
  }

  async copy(dto: CopyMenuDto) {
    const { from, to } = parsePeriod(dto.fromStart, dto.fromEnd, 92);
    const toStart = parseDay(dto.toStart);
    const source = await this.prisma.menuDay.findMany({
      where: { kind: dto.kind, date: { gte: from, lte: to } },
      select: {
        date: true,
        items: { select: { dishId: true, dish: { select: { active: true } } } },
      },
      orderBy: { date: 'asc' },
    });
    const plan = source
      .map((d) => ({
        date: copyTarget(from, toStart, d.date),
        dishIds: d.items.filter((i) => i.dish.active).map((i) => i.dishId),
      }))
      .filter((p) => p.dishIds.length);
    return this.fill(dto.kind, plan, dto.overwrite ?? false);
  }

  async deleteDay(dateValue: string, kind: GroupKind) {
    const date = parseDay(dateValue);
    const day = await this.prisma.menuDay.findUnique({
      where: { date_kind: { date, kind } },
    });
    if (!day) throw new NotFoundException('Menu day not found');
    if (day.status === MenuStatus.published)
      throw new BadRequestException('Menu is published');
    if (await this.prisma.mealOrder.count({ where: { menuDayId: day.id } }))
      throw new BadRequestException('Menu has orders');
    await this.prisma.menuDay.delete({ where: { id: day.id } });
    return { message: 'Menu day deleted' };
  }
}
```

```ts
import {
  BadRequestException,
  ConflictException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { dayRange } from '../attendance/register-calc';
import { isoWeekday } from '../timetable/schedule-calc';
import { templateWeek } from './meals-calc';
import { MenusService, parsePeriod } from './menus.service';
import {
  ApplyTemplateDto,
  CreateMenuTemplateDto,
  SaveTemplateItemsDto,
  UpdateMenuTemplateDto,
} from './dto/menu-template.dto';

const templateInclude = {
  items: {
    select: {
      week: true,
      weekday: true,
      dishId: true,
      dish: {
        select: {
          id: true,
          name: true,
          active: true,
          course: { select: { id: true, name: true, sortOrder: true } },
        },
      },
    },
    orderBy: [{ week: 'asc' as const }, { weekday: 'asc' as const }],
  },
} as const;

@Injectable()
export class MenuTemplatesService {
  constructor(
    private prisma: PrismaService,
    private menus: MenusService,
  ) {}

  list() {
    return this.prisma.menuTemplate.findMany({
      include: templateInclude,
      orderBy: { name: 'asc' },
    });
  }

  private async get(id: string) {
    const template = await this.prisma.menuTemplate.findUnique({
      where: { id },
      include: templateInclude,
    });
    if (!template) throw new NotFoundException('Menu template not found');
    return template;
  }

  async create(dto: CreateMenuTemplateDto) {
    const name = dto.name.trim();
    if (await this.prisma.menuTemplate.findUnique({ where: { name } }))
      throw new ConflictException('Menu template already exists');
    const created = await this.prisma.menuTemplate.create({
      data: { name, weeks: dto.weeks },
    });
    return this.get(created.id);
  }

  async update(id: string, dto: UpdateMenuTemplateDto) {
    const template = await this.get(id);
    const name = dto.name?.trim();
    if (
      name &&
      name !== template.name &&
      (await this.prisma.menuTemplate.findUnique({ where: { name } }))
    )
      throw new ConflictException('Menu template already exists');
    if (
      dto.weeks !== undefined &&
      (await this.prisma.menuTemplateItem.count({
        where: { templateId: id, week: { gt: dto.weeks } },
      }))
    )
      throw new BadRequestException('Template week out of range');
    await this.prisma.menuTemplate.update({
      where: { id },
      data: { name, weeks: dto.weeks, active: dto.active },
    });
    return this.get(id);
  }

  async saveItems(id: string, dto: SaveTemplateItemsDto) {
    const template = await this.get(id);
    if (dto.items.some((i) => i.week > template.weeks))
      throw new BadRequestException('Template week out of range');
    const ids = [...new Set(dto.items.map((i) => i.dishId))];
    const dishes = ids.length
      ? await this.prisma.dish.findMany({
          where: { id: { in: ids } },
          select: { id: true, active: true },
        })
      : [];
    if (dishes.length !== ids.length)
      throw new NotFoundException('Dish not found');
    const current = template.items.map((i) => i.dishId);
    if (dishes.some((d) => !d.active && !current.includes(d.id)))
      throw new BadRequestException('Dish is inactive');
    const seen = new Set<string>();
    const items = dto.items.filter((i) => {
      const key = `${i.week}:${i.weekday}:${i.dishId}`;
      if (seen.has(key)) return false;
      seen.add(key);
      return true;
    });
    await this.prisma.$transaction(async (tx) => {
      await tx.menuTemplateItem.deleteMany({ where: { templateId: id } });
      await tx.menuTemplateItem.createMany({
        data: items.map((i) => ({
          templateId: id,
          week: i.week,
          weekday: i.weekday,
          dishId: i.dishId,
        })),
      });
    });
    return this.get(id);
  }

  async apply(id: string, dto: ApplyTemplateDto) {
    const template = await this.get(id);
    if (dto.startWeek > template.weeks)
      throw new BadRequestException('Template week out of range');
    const { from, to } = parsePeriod(dto.from, dto.to, 92);
    const plan = dayRange(from, to)
      .map((date) => {
        const week = templateWeek(from, dto.startWeek, template.weeks, date);
        const weekday = isoWeekday(date);
        return {
          date,
          dishIds: template.items
            .filter(
              (i) => i.week === week && i.weekday === weekday && i.dish.active,
            )
            .map((i) => i.dishId),
        };
      })
      .filter((p) => p.dishIds.length);
    return this.menus.fill(dto.kind, plan, dto.overwrite ?? false);
  }
}
```

```ts
import { Prisma } from '@prisma/client';
import { voidChargeInTx } from '../finance/void-charge';
import { isBeforeDeadline, parseTime } from './meals-calc';
import { DEADLINE_KEY, DEFAULT_DEADLINE } from './meal-settings.service';

export async function cancelOrderInTx(
  tx: Prisma.TransactionClient,
  order: { id: string; chargeId: string },
  userId: string,
  reason: string,
) {
  await tx.mealOrder.update({
    where: { id: order.id },
    data: { cancelledAt: new Date(), cancelledById: userId, cancelReason: reason },
  });
  const charge = await tx.charge.findUnique({
    where: { id: order.chargeId },
    select: { id: true, voidedAt: true },
  });
  if (charge && !charge.voidedAt)
    await voidChargeInTx(tx, order.chargeId, reason, userId);
}

export async function cancelOrdersForNoticeInTx(
  tx: Prisma.TransactionClient,
  notice: { childId: string; startDate: Date; endDate: Date; createdAt: Date },
  userId: string,
): Promise<number> {
  await tx.$queryRaw`SELECT id FROM "Child" WHERE id = ${notice.childId} FOR UPDATE`;
  const orders = await tx.mealOrder.findMany({
    where: {
      childId: notice.childId,
      cancelledAt: null,
      date: { gte: notice.startDate, lte: notice.endDate },
    },
    select: { id: true, date: true, chargeId: true },
    orderBy: { date: 'asc' },
  });
  if (!orders.length) return 0;
  const setting = await tx.appSetting.findUnique({
    where: { key: DEADLINE_KEY },
  });
  const deadline =
    setting && parseTime(setting.value) !== null
      ? setting.value
      : DEFAULT_DEADLINE;
  let count = 0;
  for (const order of orders) {
    if (!isBeforeDeadline(order.date, deadline, notice.createdAt)) continue;
    await cancelOrderInTx(tx, order, userId, 'Відсутність');
    count++;
  }
  return count;
}
```

```ts
import {
  BadRequestException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import {
  ChargeSource,
  FinanceCategory,
  MenuStatus,
  Role,
} from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import type { AuthenticatedUser } from '../common/types/authenticated-user.type';
import { CalendarService } from '../attendance/calendar.service';
import { dayInfo } from '../attendance/register-calc';
import { parseDay } from '../classes/access';
import { isoDay, uaDay } from '../tariffs/dates';
import {
  choiceErrorMessage,
  isBeforeDeadline,
  priceOn,
  resolveChoices,
} from './meals-calc';
import { MealSettingsService } from './meal-settings.service';
import { cancelOrderInTx } from './notice-orders';
import { orderSelect, toOrderView } from './order-view';
import { OrderDayDto } from './dto/meal-order.dto';

const isParent = (user: AuthenticatedUser) => user.role === Role.parent;

const dayError = (date: Date, message: string) =>
  new BadRequestException(`${isoDay(date)}: ${message}`);

@Injectable()
export class MealOrdersService {
  constructor(
    private prisma: PrismaService,
    private calendar: CalendarService,
    private settings: MealSettingsService,
  ) {}

  private async views(ids: string[]) {
    const rows = await this.prisma.mealOrder.findMany({
      where: { id: { in: ids } },
      select: orderSelect,
      orderBy: { date: 'asc' },
    });
    return rows.map(toOrderView);
  }

  async create(
    childId: string,
    days: OrderDayDto[],
    user: AuthenticatedUser,
    now: Date = new Date(),
  ) {
    const parsed = days
      .map((d) => ({ ...d, day: parseDay(d.date) }))
      .sort((a, b) => a.day.getTime() - b.day.getTime());
    if (new Set(parsed.map((d) => d.date)).size !== parsed.length)
      throw new BadRequestException('Duplicate date in order');
    const child = await this.prisma.child.findUnique({
      where: { id: childId },
      select: {
        id: true,
        group: {
          select: {
            id: true,
            kind: true,
            mealPriceCategory: { select: { id: true, active: true } },
          },
        },
      },
    });
    if (!child) throw new NotFoundException('Child not found');
    const group = child.group;
    if (!group) throw new BadRequestException('Child has no group');
    const category = group.mealPriceCategory;
    if (!category?.active)
      throw new BadRequestException('Group has no meal price category');
    const complexIds = [...new Set(parsed.map((d) => d.complexId))];
    const [deadline, entries, complexes, prices] = await Promise.all([
      this.settings.getDeadline(),
      this.calendar.entriesBetween(
        parsed[0].day,
        parsed[parsed.length - 1].day,
      ),
      this.prisma.mealComplex.findMany({
        where: { id: { in: complexIds } },
        select: {
          id: true,
          name: true,
          active: true,
          courses: { select: { courseId: true } },
        },
      }),
      this.prisma.mealComplexPrice.findMany({
        where: { complexId: { in: complexIds }, priceCategoryId: category.id },
      }),
    ]);
    if (complexIds.some((id) => !complexes.some((c) => c.id === id)))
      throw new NotFoundException('Meal complex not found');

    const ids = await this.prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT id FROM "Child" WHERE id = ${childId} FOR UPDATE`;
      const created: string[] = [];
      for (const item of parsed) {
        const date = item.day;
        const existing = await tx.mealOrder.findFirst({
          where: { childId, date, cancelledAt: null },
          select: { id: true },
        });
        if (existing) throw dayError(date, 'Order already exists');
        if (!dayInfo(date, group, entries).working)
          throw dayError(date, 'Day is not a working day');
        const menu = await tx.menuDay.findUnique({
          where: { date_kind: { date, kind: group.kind } },
          select: { id: true, status: true },
        });
        if (!menu || menu.status !== MenuStatus.published)
          throw dayError(date, 'Menu is not published');
        if (isParent(user) && !isBeforeDeadline(date, deadline, now))
          throw dayError(date, 'Order deadline has passed');
        const complex = complexes.find((c) => c.id === item.complexId);
        if (!complex) throw new NotFoundException('Meal complex not found');
        if (!complex.active) throw dayError(date, 'Complex is inactive');
        await tx.$queryRaw`SELECT id FROM "MenuDay" WHERE id = ${menu.id} FOR SHARE`;
        const menuItems = await tx.menuDayItem.findMany({
          where: { menuDayId: menu.id },
          select: { dishId: true, dish: { select: { courseId: true } } },
        });
        const choice = resolveChoices(
          complex.courses.map((c) => c.courseId),
          menuItems.map((m) => ({ id: m.dishId, courseId: m.dish.courseId })),
          item.dishIds,
        );
        if ('error' in choice)
          throw dayError(date, choiceErrorMessage[choice.error]);
        const price = priceOn(prices, complex.id, category.id, date);
        if (!price) throw dayError(date, 'Complex has no price');
        const charge = await tx.charge.create({
          data: {
            childId,
            category: FinanceCategory.meals,
            amount: price.price,
            description: `Харчування ${uaDay(date)}: ${complex.name}`,
            periodStart: date,
            periodEnd: date,
            sourceType: ChargeSource.meals,
            createdById: user.id,
          },
        });
        const order = await tx.mealOrder.create({
          data: {
            childId,
            date,
            menuDayId: menu.id,
            complexId: complex.id,
            priceCategoryId: category.id,
            price: price.price,
            chargeId: charge.id,
            createdById: user.id,
            items: { create: choice.items },
          },
        });
        await tx.charge.update({
          where: { id: charge.id },
          data: { sourceId: order.id },
        });
        created.push(order.id);
      }
      return created;
    });
    return this.views(ids);
  }

  async cancel(
    orderId: string,
    user: AuthenticatedUser,
    reasonValue?: string,
    now: Date = new Date(),
    childId?: string,
  ) {
    const order = await this.prisma.mealOrder.findUnique({
      where: { id: orderId },
      select: {
        id: true,
        childId: true,
        date: true,
        chargeId: true,
        cancelledAt: true,
      },
    });
    if (!order || (childId && order.childId !== childId))
      throw new NotFoundException('Order not found');
    if (order.cancelledAt) throw new BadRequestException('Order is cancelled');
    const deadline = await this.settings.getDeadline();
    const before = isBeforeDeadline(order.date, deadline, now);
    const typed = reasonValue?.trim() || null;
    let reason: string;
    if (isParent(user)) {
      if (!before) throw new BadRequestException('Order deadline has passed');
      reason = 'Скасовано батьками';
    } else {
      if (!before && !typed)
        throw new BadRequestException('Reason is required');
      reason = typed ?? 'Скасовано';
    }
    await this.prisma.$transaction(async (tx) => {
      await tx.$queryRaw`SELECT id FROM "Child" WHERE id = ${order.childId} FOR UPDATE`;
      const fresh = await tx.mealOrder.findUnique({
        where: { id: orderId },
        select: { id: true, chargeId: true, cancelledAt: true },
      });
      if (!fresh) throw new NotFoundException('Order not found');
      if (fresh.cancelledAt)
        throw new BadRequestException('Order is cancelled');
      await cancelOrderInTx(tx, fresh, user.id, reason);
    });
    const [view] = await this.views([orderId]);
    if (!view) throw new NotFoundException('Order not found');
    return view;
  }
}
```

- The per-day `Meal complex not found` check inside the loop is unreachable after the pre-check but keeps TypeScript narrowing without a non-null assertion.

```ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import type { AuthenticatedUser } from '../common/types/authenticated-user.type';
import { assertGroupAccess, parseDay } from '../classes/access';
import { isoDay, kyivToday } from '../tariffs/dates';
import { orderSelect, toOrderView, type OrderView } from './order-view';
import { MealOrdersQueryDto } from './dto/meal-views.dto';

const person = { select: { id: true, firstName: true, lastName: true } };

const adminSelect = {
  ...orderSelect,
  child: {
    select: {
      id: true,
      firstName: true,
      lastName: true,
      group: { select: { id: true, name: true } },
    },
  },
  createdBy: {
    select: { id: true, firstName: true, lastName: true, role: true },
  },
  cancelledBy: person,
} as const;

export function summarize(orders: OrderView[]) {
  const complexes = new Map<
    string,
    { complexId: string; name: string; count: number }
  >();
  const dishes = new Map<
    string,
    { dishId: string; name: string; courseName: string; count: number }
  >();
  for (const order of orders) {
    if (order.cancelledAt) continue;
    const c = complexes.get(order.complex.id) ?? {
      complexId: order.complex.id,
      name: order.complex.name,
      count: 0,
    };
    c.count++;
    complexes.set(c.complexId, c);
    for (const item of order.items) {
      const d = dishes.get(item.dishId) ?? {
        dishId: item.dishId,
        name: item.dishName,
        courseName: item.courseName,
        count: 0,
      };
      d.count++;
      dishes.set(d.dishId, d);
    }
  }
  return { complexes: [...complexes.values()], dishes: [...dishes.values()] };
}

@Injectable()
export class MealViewsService {
  constructor(private prisma: PrismaService) {}

  async adminList(query: MealOrdersQueryDto) {
    const date = parseDay(query.date);
    const groupFilter =
      query.kind || query.groupId
        ? { group: { kind: query.kind, id: query.groupId } }
        : undefined;
    const rows = await this.prisma.mealOrder.findMany({
      where: { date, child: groupFilter },
      select: adminSelect,
      orderBy: [{ child: { lastName: 'asc' } }, { createdAt: 'asc' }],
    });
    const orders = rows.map((r) => ({
      ...toOrderView(r),
      child: {
        id: r.child.id,
        firstName: r.child.firstName,
        lastName: r.child.lastName,
      },
      group: r.child.group,
      createdBy: r.createdBy,
      cancelledBy: r.cancelledBy,
    }));
    return { orders, summary: summarize(orders) };
  }

  async today(
    groupId: string,
    user: AuthenticatedUser,
    today: Date = kyivToday(),
  ) {
    const group = await this.prisma.group.findUnique({
      where: { id: groupId },
      select: { id: true, teachers: { select: { id: true } } },
    });
    if (!group) throw new NotFoundException('Group not found');
    assertGroupAccess(group, user);
    const rows = await this.prisma.mealOrder.findMany({
      where: { date: today, cancelledAt: null, child: { groupId } },
      select: { ...orderSelect, child: person },
      orderBy: { child: { lastName: 'asc' } },
    });
    const views = rows.map((r) => ({ view: toOrderView(r), child: r.child }));
    return {
      date: isoDay(today),
      orders: views.map(({ view, child }) => ({
        id: view.id,
        child,
        complex: view.complex,
        items: view.items,
      })),
      summary: summarize(views.map((v) => v.view)),
    };
  }
}
```

```ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { MenuStatus } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';
import type { AuthenticatedUser } from '../common/types/authenticated-user.type';
import { assertOwnChild } from '../common/ownership';
import { CalendarService } from '../attendance/calendar.service';
import { dayInfo } from '../attendance/register-calc';
import { formatMoney } from '../finance/finance-calc';
import { isoDay } from '../tariffs/dates';
import {
  deadlineFor,
  isBeforeDeadline,
  priceOn,
  resolveChoices,
} from './meals-calc';
import { MealSettingsService } from './meal-settings.service';
import { MealOrdersService } from './meal-orders.service';
import { parsePeriod } from './menus.service';
import { orderSelect, toOrderView } from './order-view';
import { ParentMealOrdersDto } from './dto/meal-order.dto';

const byCourse = <
  T extends { name: string; course: { sortOrder: number; name: string } },
>(
  a: T,
  b: T,
) =>
  a.course.sortOrder - b.course.sortOrder ||
  a.course.name.localeCompare(b.course.name) ||
  a.name.localeCompare(b.name);

@Injectable()
export class ParentMealsService {
  constructor(
    private prisma: PrismaService,
    private calendar: CalendarService,
    private settings: MealSettingsService,
    private orders: MealOrdersService,
  ) {}

  async menu(
    parentId: string,
    childId: string,
    fromValue: string,
    toValue: string,
    now: Date = new Date(),
  ) {
    await assertOwnChild(this.prisma, parentId, childId);
    const { from, to } = parsePeriod(fromValue, toValue, 31);
    const deadline = await this.settings.getDeadline();
    const child = await this.prisma.child.findUnique({
      where: { id: childId },
      select: {
        id: true,
        group: {
          select: {
            id: true,
            kind: true,
            mealPriceCategory: { select: { id: true, active: true } },
          },
        },
      },
    });
    const group = child?.group;
    if (!group) return { deadline, days: [] };
    const category = group.mealPriceCategory?.active
      ? group.mealPriceCategory
      : null;
    const [entries, menus, complexes, prices, orders] = await Promise.all([
      this.calendar.entriesBetween(from, to),
      this.prisma.menuDay.findMany({
        where: {
          kind: group.kind,
          status: MenuStatus.published,
          date: { gte: from, lte: to },
        },
        select: {
          id: true,
          date: true,
          items: {
            select: {
              dish: {
                select: {
                  id: true,
                  name: true,
                  course: { select: { id: true, name: true, sortOrder: true } },
                },
              },
            },
          },
        },
        orderBy: { date: 'asc' },
      }),
      category
        ? this.prisma.mealComplex.findMany({
            where: { active: true },
            select: {
              id: true,
              name: true,
              description: true,
              courses: {
                select: {
                  course: { select: { id: true, name: true, sortOrder: true } },
                },
              },
            },
            orderBy: [{ sortOrder: 'asc' }, { name: 'asc' }],
          })
        : Promise.resolve([]),
      category
        ? this.prisma.mealComplexPrice.findMany({
            where: { priceCategoryId: category.id },
          })
        : Promise.resolve([]),
      this.prisma.mealOrder.findMany({
        where: {
          childId,
          cancelledAt: null,
          date: { gte: from, lte: to },
        },
        select: orderSelect,
      }),
    ]);
    const days = menus
      .filter((m) => dayInfo(m.date, group, entries).working)
      .sort((a, b) => a.date.getTime() - b.date.getTime())
      .map((m) => {
        const dishes = m.items.map((i) => i.dish).sort(byCourse);
        const open = isBeforeDeadline(m.date, deadline, now);
        const available = category
          ? complexes.flatMap((c) => {
              const price = priceOn(prices, c.id, category.id, m.date);
              const courses = c.courses
                .map((cc) => cc.course)
                .sort(
                  (a, b) =>
                    a.sortOrder - b.sortOrder || a.name.localeCompare(b.name),
                );
              const check = resolveChoices(
                courses.map((cc) => cc.id),
                dishes.map((d) => ({ id: d.id, courseId: d.course.id })),
                [],
              );
              if (!price || ('error' in check && check.error === 'course_missing'))
                return [];
              return [
                {
                  id: c.id,
                  name: c.name,
                  description: c.description,
                  price: formatMoney(price.price),
                  courses: courses.map((cc) => ({
                    id: cc.id,
                    name: cc.name,
                    options: dishes
                      .filter((d) => d.course.id === cc.id)
                      .map((d) => ({ id: d.id, name: d.name })),
                  })),
                },
              ];
            })
          : [];
        const order = orders.find((o) => o.date.getTime() === m.date.getTime());
        return {
          date: isoDay(m.date),
          deadlineAt: deadlineFor(m.date, deadline).toISOString(),
          open,
          dishes,
          complexes: open ? available : [],
          order: order ? toOrderView(order) : null,
          canOrder: open && !order && available.length > 0,
          canCancel: open && !!order,
        };
      });
    return { deadline, days };
  }

  async order(user: AuthenticatedUser, childId: string, dto: ParentMealOrdersDto) {
    await assertOwnChild(this.prisma, user.id, childId);
    return this.orders.create(childId, dto.days, user);
  }

  async cancel(user: AuthenticatedUser, childId: string, orderId: string) {
    await assertOwnChild(this.prisma, user.id, childId);
    const view = await this.orders.cancel(
      orderId,
      user,
      undefined,
      new Date(),
      childId,
    );
    if (!view) throw new NotFoundException('Order not found');
    return view;
  }
}
```

```ts
export const translateError = (error: unknown): string => {
  if (typeof error === 'object' && error !== null && 'message' in error) {
    const { message } = error as { message: unknown };
    if (typeof message === 'string') {
      const dated = /^(\d{4})-(\d{2})-(\d{2}): (.+)$/.exec(message);
      if (dated) {
        const text = errorMessages[dated[4]];
        return text ? `${dated[3]}.${dated[2]}.${dated[1]}: ${text}` : GENERIC_ERROR;
      }
      return errorMessages[message] ?? GENERIC_ERROR;
    }
  }
  return GENERIC_ERROR;
};
```

```ts
          mealsApi.getSettings(),
        ]);
        set({
          courses: courses.data,
          dishes: dishes.data,
          priceCategories: categories.data,
          complexes: complexes.data,
          deadline: settings.data.mealOrderDeadline,
        });
      } catch (error) {
        set({ loadFailed: true });
        throw error;
      }
    },
    saveCourse: (id, dto) =>
      saving(async () => {
        const { data } = id ? await mealsApi.updateCourse(id, dto) : await mealsApi.createCourse(dto);
        set((s) => ({ courses: bySort(replace(s.courses, data)) }));
      }),
    saveDish: (id, dto) =>
      saving(async () => {
        const { data } = id ? await mealsApi.updateDish(id, dto) : await mealsApi.createDish(dto);
        set((s) => ({ dishes: replace(s.dishes, data) }));
      }),
    savePriceCategory: (id, dto) =>
      saving(async () => {
        const { data } = id ? await mealsApi.updatePriceCategory(id, dto) : await mealsApi.createPriceCategory(dto);
        set((s) => ({ priceCategories: bySort(replace(s.priceCategories, data)) }));
      }),
    saveComplex: (id, dto) =>
```

```tsx
  const kind = child?.group?.kind
  const menuKey = kind && date ? `${kind}/${date}` : null

    setMenu({ key: k, day: null, failed: false })
    mealsApi
      .getMenuDays(groupKind, day, day)
      .then(({ data }) => setMenu((m) => (m?.key === k ? { key: k, day: data[0] ?? null, failed: false } : m)))
```

```ts
import { create } from 'zustand';
import { addDaysIso, localToday, mealsApi, weekStart, type OrderDay, type ParentMenu } from '../../../shared/api';

interface ChildMealsState {
  week: string;
  key: string | null;
  data: ParentMenu | null;
  loadFailed: boolean;
  isSaving: boolean;
  load: (childId: string, week?: string) => Promise<void>;
  order: (childId: string, days: OrderDay[]) => Promise<void>;
  cancel: (childId: string, orderId: string) => Promise<void>;
}

export const useChildMealsStore = create<ChildMealsState>((set, get) => {
  const saving = async (childId: string, fn: () => Promise<unknown>) => {
    set({ isSaving: true });
    try {
      await fn();
      await get().load(childId);
    } finally {
      set({ isSaving: false });
    }
  };
  return {
    week: weekStart(localToday()),
    key: null,
    data: null,
    loadFailed: false,
    isSaving: false,
    load: async (childId, week = get().week) => {
      const key = `${childId}/${week}`;
      set({ week, key, data: get().key === key ? get().data : null, loadFailed: false });
      try {
        const { data } = await mealsApi.getParentMenu(childId, week, addDaysIso(week, 6));
        if (get().key === key) set({ data });
      } catch (error) {
        if (get().key === key) set({ loadFailed: true });
        throw error;
      }
    },
    order: (childId, days) => saving(childId, () => mealsApi.parentOrder(childId, days)),
    cancel: (childId, orderId) => saving(childId, () => mealsApi.parentCancel(childId, orderId)),
  };
});
```
